\chapter*{Basic of EM waves}
\begin{question}
    Describe the Maxwell  equations
\end{question}

\begin{answer}
    % Legge di Faraday
    $\nabla \times \bar{e} = -\frac{\partial \bar{b}}{\partial t}$

    % Legge di Gauss (elettrica)
    $\nabla \cdot \bar{d} = \rho$

    % Legge di Ampère-Maxwell
    $\nabla \times \bar{h} = \bar{j} + \frac{\partial \bar{d}}{\partial t}$

    % Legge di Gauss (magnetica)
    $\nabla \cdot \bar{b} = 0$

    \begin{itemize}
        \item $\bar{e}(\bar{r}, t)$ is ELECTRIC FIELD $[V/m]$
        \item $\bar{d}(\bar{r}, t)$ is the electric flux density $[C/m^2]$
        \item $\bar{h}(\bar{r}, t)$ is MAGNETIC FIELD $[A/m]$
        \item $\bar{b}(\bar{r}, t)$ is MAGNETIC FLUX DENSITY $[T = Wb/m^2]$
        \item $\rho(\bar{r}, t)$ is CHARGE DENSITY $[C/m^3]$
        \item $\bar{J}(\bar{r}, t)$ is ELECTRIC CURRENT DENSITY $[A/m^2]$
        \item $\bar{r} = (x, y, z)$
    \end{itemize}
    Assumptions: $\rho = 0, \vec{J} = 0$
\end{answer}

\begin{question}
    EM fields in harmonic domain
\end{question}

\begin{answer}
    In many cases we can limit the study of EM fields to the harmonic signals. It is convenient to represent fields by their phaser using the following relation:
    \begin{equation}
        \bar{e}(\bar{r}, t) = \text{Re}\left[ \bar{E}(\bar{r}) e^{j\omega t} \right]
    \end{equation}
    In this case we have no electric charges or current. Hence:
    \begin{align*}
        \nabla \times \bar{E} = -j\omega \bar{B} = -j\omega \mu \bar{H}
    \end{align*}
    \begin{align*}
        \nabla \cdot \bar{D} = 0
    \end{align*}
    \begin{align*}
        \nabla \times \bar{H} = j\omega \bar{D} = j\omega \epsilon \bar{E}
    \end{align*}
    \begin{align*}
        \nabla \cdot \bar{B} = 0
    \end{align*}
\end{answer}

\begin{question}
    Describe the constitutive relations
\end{question}

\begin{answer}
    The relation between fields and flux densities depend on the medium where the fields propagate. In general we have:
    \begin{align*}
        \bar{D} = \epsilon \bar{E}
    \end{align*}
    \begin{align*}
        \bar{B} = \mu \bar{H}
    \end{align*}
    where $\epsilon$ is the electric permittivity and $\mu$ is the magnetic permeability of the medium. $[F/m]$ and $[H/m]$ respectively.
\end{answer}

\begin{question}
    Describe Helmholtz equation
\end{question}

\begin{answer}
    The Helmholtz equation is a second-order partial differential equation that describes the propagation of waves in a medium.
    Starting from the Maxwell equations in harmonic domain without charges and currents, we can derive the Helmholtz equation for the electric and magnetic fields.
    \begin{gather*}
        \text{\textbf{Maxwell's Equations (Phasor Form)}} \\
        \begin{aligned}
            \nabla \times \overline{E} & = -j\omega\mu \overline{H}        & \nabla \cdot \overline{E} & = 0 \\
            \nabla \times \overline{H} & = j\omega\varepsilon \overline{E} & \nabla \cdot \overline{H} & = 0
        \end{aligned}
    \end{gather*}

    \begin{gather*}
        \text{\textbf{Derivation of the Helmholtz Equation}} \\
        \text{1) Apply the curl to Faraday's Law:} \\
        \nabla \times (\nabla \times \overline{E}) = \nabla \times (-j\omega\mu \overline{H}) \\
        \\
        \text{2) Due to the linearity of the operator, move the constants outside:} \\
        \nabla \times (\nabla \times \overline{E}) = -j\omega\mu (\nabla \times \overline{H}) \\
        \\
        \text{3) Substitute the curl of } \overline{H} \text{ using the Ampère-Maxwell Law:} \\
        \nabla \times (\nabla \times \overline{E}) = -j\omega\mu (j\omega\varepsilon \overline{E}) \\
        \\
        \text{4) Simplify the constant terms (given that } j^2 = -1\text{):} \\
        \nabla \times (\nabla \times \overline{E}) = \omega^2\mu\varepsilon \overline{E} \\
        \\
        \text{5) Use the vector identity } \nabla \times \nabla \times \overline{A} = \nabla(\nabla \cdot \overline{A}) - \nabla^2\overline{A} \text{:} \\
        \nabla(\nabla \cdot \overline{E}) - \nabla^2\overline{E} = \omega^2\mu\varepsilon \overline{E} \\
        \\
        \text{Since } \nabla \cdot \overline{E} = 0 \text{ in a source-free medium, the gradient term vanishes:} \\
        -\nabla^2\overline{E} = \omega^2\mu\varepsilon \overline{E} \\
        \\
        \text{6) Rearrange the terms to obtain the Helmholtz Equation:} \\
        \nabla^2\overline{E} + \omega^2\mu\varepsilon \overline{E} = 0
    \end{gather*}

    For magnetic field the steps are the same.

\end{answer}

\begin{question}
    What is the advantage of using piecewise homogeneous media?
\end{question}

\begin{answer}
    Piecewise homogeneous is a semplification that allows us to consider $\epsilon$ and $\mu$ as constants in a given region.
    But normally $\epsilon$ and $\mu$ are not constant, so we solve this problem by using boundary conditions.
\end{answer}

\begin{question}
    What are the boundary conditions?
\end{question}

\begin{answer}
    We consider the thin surface between 2 different materials. We define the normal versor $\hat{n}$ .
    \subsection*{For flux densities}
    \begin{itemize}
        \item \textbf{Density of magnetic flux}: normal component is always continuous
              \begin{equation}
                  \hat{n} \cdot (\bar{B}_2 - \bar{B}_1) = 0
              \end{equation}
        \item \textbf{Density of electric flux}: normal component is discontinuous
              \begin{equation}
                  \hat{n} \cdot (\bar{D}_2 - \bar{D}_1) = \rho_s
              \end{equation}
    \end{itemize}
    \subsection*{For fields}
    \begin{itemize}
        \item \textbf{Electric field}: tangential component is always continuous
              \begin{equation}
                  \hat{n} \times (\bar{E}_2 - \bar{E}_1) = 0
              \end{equation}
        \item \textbf{Magnetic field}: tangential component is discontinuous
              \begin{equation}
                  \hat{n} \times (\bar{H}_2 - \bar{H}_1) = \bar{J}_s
              \end{equation}
    \end{itemize}

    For our assumptions, $\rho_s = 0$ and $\bar{J}_s = 0$.
\end{answer}

\begin{question}
    Describe the Poynting vector
\end{question}

\begin{answer}
    In the time domain the Poynting vector is defined as:
    \begin{equation}
        \bar{P}(\bar{r}, t) = \bar{E}(\bar{r}, t) \times \bar{H}(\bar{r}, t)
    \end{equation}
    where $\bar{E}$ is the electric field and $\bar{H}$ is the magnetic field.\\
    While in the harmonic domain we have:
    \begin{equation}
        \bar{P}(\bar{r}) = \frac{1}{2} \bar{E}(\bar{r}) \times \bar{H}^*(\bar{r})
    \end{equation}
    where $\bar{H}^*$ is the complex conjugate of $\bar{H}$.\\
    In other words, it represents the power flow per unit area.

\end{answer}

\begin{question}
    Describe the polarization of EM waves and the polarization plane
\end{question}

\begin{answer}
    A generic sinusoidal EM field can be written as:
    \begin{equation}
        \overline{e}(\overline{r}, t) = \sum_{n=1}^{3 \, \text{dim}} C_n(\overline{r}) \cos(\omega t + \phi_n(\overline{r})) \hat{x}_n = \underbrace{c'(\overline{r}, t)}_{\overline{E}_{\text{Re}}} \cos(\omega t) - \underbrace{c''(\overline{r})}_{\overline{E}_{\text{Im}}} \sin(\omega t)
    \end{equation}
    which clearly shows that the electric vector of a generic sinusoidal field lays at any time on a plane defined by the vectors $\overline{E}_{\text{Re}}$ and $\overline{E}_{\text{Im}}$.

    This is known as polarization plane: constant in time, vary in space.
    The sinusoidal field is periodic, so the curve it describes must be closed. Generally, it is an ellipse.

    This ellipse is described by two components:
    \begin{align}
        \bar{e}'(t) =
        \begin{bmatrix} a \cos(\omega t + \phi) \\ b \sin(\omega t + \phi)
        \end{bmatrix}
    \end{align}

    We that the polarization is:
    \begin{itemize}
        \item Right handed: when the field rotates counterclockwise
        \item Left handed: when the field rotates clockwise
    \end{itemize}
    Two notable cases are:
    \begin{itemize}
        \item Linear polarization: when the ellipse degenerates into a line
        \item Circular polarization: when the ellipse degenerates into a circle ($a=\pm b$)
    \end{itemize}
\end{answer}

\begin{question}
    Describe Jones vector and their utility
\end{question}

\begin{answer}
    It's a bidimensional complex vector which describes a generic sinusoidal EF when represented with respect to its polarization plane.
    The Jones vector reads:
    \begin{equation}
        \bar{J} =
        \begin{bmatrix} a\cos(\delta) + j b \sin(\delta) \\ a\sin(\delta) - j b \cos(\delta)
        \end{bmatrix}
    \end{equation}

    \begin{itemize}
        \item $\Phi$ does not influence the polarization, but it's useful to describe field propagation;
        \item Two polarizations are orthogonal if the corresponding Jones vectors are orthogonal. ($\bar{E}_1 \cdot \bar{E}_2 = 0$)
    \end{itemize}
\end{answer}

\begin{question}
    Describe Stokes vector
\end{question}

\begin{answer}
    Stokes vectors are real vectors which describes a more intuitive geometrical representation of light polarization.
    The Stokes vector reads:
    \begin{equation}
        \vec{S} =
        \begin{bmatrix} S_0 \\ S_1 \\ S_2 \\ S_3
        \end{bmatrix} =
        \begin{bmatrix}
            |E_1|^2 + |E_2|^2       \\
            |E_1|^2 - |E_2|^2       \\
            2 \text{Re} [E_1 E_2^*] \\
            2 \text{Im} [E_1 E_2^*]
        \end{bmatrix} =
        \begin{bmatrix}
            a^2 + b^2                \\
            (a^2 - b^2) \cos 2\delta \\
            (a^2 - b^2) \sin 2\delta \\
            2ab
        \end{bmatrix}
    \end{equation}

    \begin{itemize}
        \item $S_0$ total intensity;
        \item $S_1$ difference in intensities between horizontal and vertical linearly polarized components;
        \item $S_2$ difference in intensities between linearly polarized components oriented at +45° and -45°;
        \item $S_3$ difference in intensities between right and left circularly polarized components
    \end{itemize}
    It does not depend on phase.
    \begin{align}
        S_0 = \sqrt{S_1^2 + S_2^2 + S_3^2}
    \end{align}
    Monochromatic case: $\omega$ is constant.\\
    Also, knowing the redundancy of $S_0$, we can define the unit Stokes vector:
    \begin{align}
        \hat{S} = \begin{bmatrix} S_1 \\ S_2 \\ S_3 \end{bmatrix} =
        \frac{1}{S_0} \begin{bmatrix} S_1 \\ S_2 \\ S_3 \end{bmatrix} =
        \frac{1}{a^2 + b^2} \begin{bmatrix}
                                (a^2 - b^2) \cos 2\delta \\
                                (a^2 - b^2) \sin 2\delta \\
                                2ab
                            \end{bmatrix}
    \end{align}

    which can be further simplified with ellipticity: $\epsilon = \tan^{-1}(b/a)$


    \begin{align}
        % Definizione del parametro di ellitticità
        \tan \epsilon  & = \frac{b}{a}                                                                                                     \\
        % Identità trigonometriche per l'angolo doppio
        \cos 2\epsilon & = \frac{1 - \tan^2 \epsilon}{1 + \tan^2 \epsilon} = \frac{1 - (b/a)^2}{1 + (b/a)^2} = \frac{a^2 - b^2}{a^2 + b^2} \\
        \sin 2\epsilon & = \frac{2 \tan \epsilon}{1 + \tan^2 \epsilon} = \frac{2(b/a)}{1 + (b/a)^2} = \frac{2ab}{a^2 + b^2}                \\
        % Sostituzione nel vettore unitario s_hat
        \hat{s}        & = \frac{1}{a^2 + b^2} \begin{bmatrix} (a^2 - b^2) \cos 2\delta \\ (a^2 - b^2) \sin 2\delta \\ 2ab \end{bmatrix}   \\
        % Risultato finale in coordinate angolari (Sfera di Poincaré)
        \hat{s}        & = \begin{bmatrix} \cos 2\epsilon \cos 2\delta \\ \cos 2\epsilon \sin 2\delta \\ \sin 2\epsilon \end{bmatrix}
    \end{align}
\end{answer}

        \item North pole: circular polarization right handed

\chapter*{Modal theory}
\begin{question}
    What is Total Internal Reflection?
\end{question}

\begin{answer}
    Consider a fiber. We have the core and the cladding. We have to respect two conditions:
    \begin{itemize}
        \item The refractive index of the incidence medium must be greater than the refractive index of the transmission medium ($n_1 > n_2$);
        \item The angle of incidence $\theta_i$ must be greater than the critical angle $\theta_c = \sin^{-1}(n_2/n_1)$.
    \end{itemize}

    When $\theta_i < \theta_c$ we have \textit{uniform trasmitted waves} and $n_1 \sin(\theta_i) = n_2 \sin(\theta_t)$.

\end{answer}


\begin{question}
    Considered a step-index fiber; let us determine the maximum input angle that a ray
    may have so that the ray refracted in the fiber core is guided by TIR.
\end{question}

\begin{answer}
    \begin{itemize}
        \item \textbf{Snell's Law at the input interface (Air to Core):}
              \begin{equation}
                  n_0 \sin \beta_M = n_1 \sin \beta_l \implies \sin \beta_M = n_1 \sqrt{1 - \cos^2 \beta_l} \quad (\text{assuming } n_0 \approx 1)
              \end{equation}

        \item \textbf{Total Internal Reflection (TIR) condition at Core-Cladding interface:}
              The ray is guided if the incidence angle is greater than or equal to the critical angle $\theta_c$:
              \begin{equation}
                  \frac{\pi}{2} - \beta_l \ge \theta_c \implies \cos \beta_l \ge \sin \theta_c
              \end{equation}

        \item \textbf{Critical Angle definition:}
              \begin{equation}
                  \sin \theta_c = \frac{n_2}{n_1}
              \end{equation}

        \item \textbf{Substitution for Maximum Angle ($\beta_M$):}
              Substituting the limiting case $\cos \beta_l = \frac{n_2}{n_1}$ into the first equation:
              \begin{equation}
                  \sin \beta_M = n_1 \sqrt{1 - \left( \frac{n_2}{n_1} \right)^2} = \sqrt{n_1^2 - n_2^2}
              \end{equation}

        \item \textbf{Final Result (Numerical Aperture):}
              \begin{equation}
                  NA = \sin \beta_M = \sqrt{n_1^2 - n_2^2}
              \end{equation}
    \end{itemize}

    Large NA $\implies$ easier to couple light into the fiber, but can cause ISI (modal dispersion).

\end{answer}

\begin{question}
    Estimate modal dispersion in a step-index fiber.
\end{question}

\begin{answer}
    \begin{align}
        % 1. Definizione delle lunghezze dei percorsi
        T_1               & = L \quad (\text{fiber length})                                                                                                                              \\
        T_2               & = \frac{L}{\cos \beta_l}                                                                                                                                     \\
        % 2. Velocità di propagazione nel core
        v                 & = \frac{c}{n_1} \quad (\text{speed of propagation})                                                                                                          \\
        % 3. Differenza di ritardo temporale (Delay spread)
        \Delta \tau       & = \frac{\Delta L}{v} = L \left( \frac{1}{\cos \beta_l} - 1 \right) \frac{n_1}{c}                                                                             \\
        % 4. Vincolo per la riflessione totale interna
                          & \text{Nonetheless, it must be: } \cos \beta_l \geq \frac{n_2}{n_1}                                                                                           \\
        % 5. Ritardo massimo (Worst-case scenario)
        \Delta \tau       & = \frac{\Delta L}{v} = L \left( \frac{1}{\cos \beta_l} - 1 \right) \frac{n_1}{c} \leq L \left( \frac{n_1}{n_2} - 1 \right) \frac{n_1}{c} = \Delta \tau_{max} \\
        % 6. Approssimazione per fibre a debole guida (n1 \approx n2)
                          & \text{If we assume that } n_1 \approx n_2 \implies NA^2 \approx 2n_1(n_1 - n_2)                                                                              \\
        % 7. Formula finale per il ritardo massimo
        \Delta \tau_{max} & \approx L \frac{NA^2}{2 n_2 c} \quad \text{(max delay the signal can experience)}
    \end{align}
\end{answer}

\begin{question}
    What is Graded-Index Fiber?
\end{question}

\begin{answer}
    A graded-index (GRIN) fiber is a type of optical fiber, typically multimode, designed specifically to reduce intermodal dispersion by managing how light rays travel through the core.\\
    The refractive index $n(r)$ varies radially. It is highest at the fiber axis ($n_1$) and lowest at the core-cladding interface ($n_2$). This variation is often modeled by a power law (the $\alpha$-profile), which is typically parabolic ($\alpha \approx 2$):
    $$
        n^{2}(r)=n_{1}^{2}\left[1-2 \Delta\left(\frac{r}{a}\right)^{g}\right]
    $$
    where:
    \begin{itemize}
        \item $n_1$ is the refractive index at the fiber axis;
        \item $n_2$ is the refractive index of the cladding;
        \item $\Delta = \frac{n_1 - n_2}{n_1}$ is the relative refractive index difference;
        \item $a$ is the radius of the core;
        \item $g$ is the profile parameter (typically $g \approx 2$ for parabolic profile).
    \end{itemize}
\end{answer}

\begin{question}
    What are the rays type?
\end{question}

\begin{answer}
    \begin{itemize}
        \item Ray Type 1: shortest path, highest refractive index, lowest propagation speed
        \item Ray Type 2: longest path, lowest refractive index, average propagation speed
        \item Ray Type 3: helical path, lowest refractive index, highest propagation speed
    \end{itemize}
\end{answer}

Also, when $g=2$ we can calculate the maximum intermodal dispersion, that is:
\begin{equation}
    \Delta \tau_{max} \approx \frac{L}{32 c n_2} \dot {NA}^4
\end{equation}
which for a small NA (less than 1), we have a smaller intermodal dispersion.

\begin{question}
    Describe Guided modes and radiation modes.
\end{question}

\begin{answer}
    \begin{itemize}
        \item \textbf{Guided Modes}: These are field structures that are confined within the fiber's core and guided by it. They are the desired modes for transmitting power.
        \item \textbf{Radiation Modes}: These are field structures that are not confined by the guiding structure of the fiber. Their excitation represents a loss of power and should be avoided.
    \end{itemize}
\end{answer}

    Step index fiber propagation modes.
\end{question}

\begin{answer}
    General modes:
    \begin{itemize}
        \item Transverse electric ($TE_{0,p}$)
        \item Transverse magnetic ($TM_{0,p}$)
        \item Hybrid electric ($HE_{m,p}$)
        \item Hybrid magnetic ($HM_{m,p}$)
    \end{itemize}

    First higher order mode: $TE_{0,1}$ and $TM_{0,1}$\\
    Weak guide approximation: $n_{core} \approx n_{cladding}$ \\
    LP modes: Linearly polarized modes
\end{answer}

\begin{question}
    Modal theory
\end{question}

\begin{answer}
    EM field decomposed into \textbf{transverse and longitudinal components}
    \begin{equation*}
        \boxed{
            \begin{cases}
                \bar{E}(x,y,z) = \left\{ \bar{E}_t(x,y) + E_z(x,y)\hat{z} \right\} e^{-j\beta z} \\
                \bar{H}(x,y,z) = \left\{ \bar{H}_t(x,y) + H_z(x,y)\hat{z} \right\} e^{-j\beta z}
            \end{cases}
        }
    \end{equation*}
    \begin{itemize}
        \item Propagation constant $\Rightarrow \beta$ unknown
    \end{itemize}

    \vspace{1em}
    \noindent If we insert the previous expressions in Maxwell's eq. we have:

    \begin{itemize}
        \item[\textbf{[1]}] $\nabla_t^2 \bar{E}_t - (\beta^2 - k^2) \bar{E}_t = 0 \Rightarrow$ Helmholtz eq. for transverse component of $E_t$

        \item[\textbf{[2]}] $j \beta E_z = \nabla \cdot \bar{E}_t \Rightarrow$ This connects the longitudinal component of the electric field ($E_z$) to the divergence of transversal component of e.f. ($\nabla \cdot \bar{E}_t$)

        \item[\textbf{[3a]}] $j \omega \mu_o \bar{H}_t = \hat{z} \times (\nabla E_z + j \beta \bar{E}_t) \Rightarrow$ relation between transverse comp. of $H_t$ with electric field components and their derivatives

        \item[\textbf{[3b]}] $j \beta H_z = \nabla \cdot \bar{H}_t \Rightarrow$ relation between long. component of $H_z$ and the divergence of transv. component of magnetic field ($\nabla \cdot \bar{H}_t$)
    \end{itemize}
    where:
    \begin{equation*}
        k^2 =
        \begin{cases}
            \omega^2 \mu_0 \varepsilon_{co} = k_{co}^2 & \text{in the core}     \\
            \omega^2 \mu_0 \varepsilon_{cl} = k_{cl}^2 & \text{in the cladding}
        \end{cases}
    \end{equation*}
    Also, the eigenvalue are defined as:
    \begin{align*}
        \chi_{co}^2 & = k_{co}^2 - \beta^2 = \omega^2 \mu_0 \epsilon_{co} - \beta^2 \\
        \chi_{cl}^2 & = k_{cl}^2 - \beta^2 = \omega^2 \mu_0 \epsilon_{cl} - \beta^2
    \end{align*}
    \noindent It can be shown that the propagation constant must fulfill the following important constraint:
    \begin{equation*}
        k_{cl} = \omega \sqrt{\mu_0 \epsilon_{cl}} < \beta < \omega \sqrt{\mu_0 \epsilon_{co}} = k_{co}
    \end{equation*}

    \noindent therefore $\chi_{co}^2 \geq 0$ and $\chi_{cl}^2 \leq 0$.

    \vspace{1em}
    \noindent It is convenient to define the following real quantities:
    \begin{equation*}
        \chi_{co} = \sqrt{k_{co}^2 - \beta^2}, \quad \quad \tilde{\chi}_{cl} = \sqrt{\beta^2 - k_{cl}^2}
    \end{equation*}

    \noindent Notice that:
    \begin{equation*}
        \chi_{co}^2 - \chi_{cl}^2 = \chi_{co}^2 + \tilde{\chi}_{cl}^2 = \left( \frac{2\pi}{\lambda} \right)^2 \text{NA}^2
    \end{equation*}

    we can also say that $\chi_\text{core}^2 \geq 0, \chi_\text{cladding}^2 \leq 0 \Rightarrow \chi_\text{cladding} \in \mathbb{C}$\\
    Now we have to solve the Helmholtz equation:
    $$\nabla^2\vec{E_t} + \chi^2\vec{E_t} = 0 \qquad \vec{E_t} = \left[\begin{matrix}
                E_x \\
                E_y \\
                0
            \end{matrix}\right]$$
    Here comes the LP assumption $\Rightarrow$ polarization stays the same everywhere on the cross section, hence:
    $$\text{Y-polarization} \qquad \Rightarrow \qquad E_x = 0 \qquad \Rightarrow \qquad \vec{E_t} = \left[\begin{matrix}
                0   \\
                E_y \\
                0
            \end{matrix}\right]$$
\end{answer}

\begin{question}
    Explain briefly effective refractive index
\end{question}

\begin{answer}
    The effective refractive index ($n_{\text{eff}}$ or $\bar{n}$) is a parameter that quantifies the phase velocity of a specific mode propagating within an optical fiber. It connects the propagation constant $\beta$ of the mode to the frequency of light $\omega$ and the speed of light $c$.

    \begin{itemize}
        \item \textbf{Definition:} It is defined by the relationship:
              \begin{equation*}
                  \beta = \frac{\omega}{c} n_{\text{eff}} \quad \left( \text{or equivalently } n_{\text{eff}} = \frac{\beta}{k_0} \right)
              \end{equation*}

        \item \textbf{Physical Interpretation:} It can be interpreted as the average refractive index ``seen'' by the optical field of a guided mode as it propagates. Since the optical field extends across both the core and the cladding, $n_{\text{eff}}$ represents a weighted average of the refractive indices of these two regions.

        \item \textbf{Range:} For a guided mode, the effective refractive index is always bounded between the refractive index of the cladding ($n_{\text{cl}}$ or $n_2$) and the core ($n_{\text{co}}$ or $n_1$):
              \begin{equation*}
                  n_{\text{cl}} < n_{\text{eff}} < n_{\text{co}}
              \end{equation*}

        \item \textbf{Cut-off Condition:} $n_{\text{eff}}$ depends on the frequency (wavelength); if $n_{\text{eff}}$ decreases until it equals $n_{\text{cl}}$, the mode ceases to be guided and is radiated into the cladding (cut-off condition).

        \item \textbf{Normalization:} It is often expressed via the normalized propagation constant $b$, where:
              \begin{equation*}
                  b \approx \frac{n_{\text{eff}} - n_{\text{cl}}}{n_{\text{co}} - n_{\text{cl}}}
              \end{equation*}
              which ranges from 0 to 1.
    \end{itemize}
\end{answer}

\begin{question}
    Sketch the way in which LP modes can be calculated
\end{question}

\begin{answer}

    The \textbf{Weakly Guiding Approximation} assumes the refractive index difference between core and cladding is very small:
    \begin{equation}
        \Delta \approx \frac{(NA)^2}{2n_{cl}^2} \ll 1 \implies n_{co} \approx n_{cl}
    \end{equation}

    Under these conditions, the transverse components of the electric field ($E_x, E_y$) satisfy the scalar Helmholtz equation in cylindrical coordinates $\{r, \phi, z\}$:
    \begin{equation}
        \left[ \frac{\partial^2}{\partial r^2} + \frac{1}{r}\frac{\partial}{\partial r} + \frac{1}{r^2}\frac{\partial^2}{\partial \phi^2} + (\omega^2\mu\epsilon - \beta^2) \right] \vec{E}_t = 0
    \end{equation}

    \noindent Using separation of variables ($E_t = f(r)g(\phi)$), the solutions are expressed via Bessel functions of order $n$:
    \begin{itemize}
        \item \textbf{Core ($r < a$):} $E_{y,co} = q \frac{J_n(\chi_{co}r)}{J_n(\chi_{co}a)} \cos(n\phi + \phi_0)$ (Oscillatory)
        \item \textbf{Cladding ($r \geq a$):} $E_{y,cl} = q \frac{K_n(\tilde{\chi}_{cl}r)}{K_n(\tilde{\chi}_{cl}a)} \cos(n\phi + \phi_0)$ (Evanescent)
    \end{itemize}
    where $\chi_{co} = \sqrt{k_0^2 n_{co}^2 - \beta^2}$ and $\tilde{\chi}_{cl} = \sqrt{\beta^2 - k_0^2 n_{cl}^2}$.

    \noindent Applying boundary conditions at $r=a$ yields the eigenvalue equation for $LP_{n,p}$ modes:
    \begin{equation}
        (\chi_{co} a) \frac{J_{n+1}(\chi_{co}a)}{J_{n}(\chi_{co}a)} = (\tilde{\chi}_{cl} a) \frac{K_{n+1}(\tilde{\chi}_{cl}a)}{K_{n}(\tilde{\chi}_{cl}a)}
    \end{equation}

    \textbf{Physical Constraints:}
    \begin{itemize}
        \item \textbf{Guidance Condition:} $n_{cl} < n_{eff} < n_{co}$ (where $n_{eff} = \beta/k_0$).
        \item \textbf{Cut-off:} Occurs when $\beta \to k_0 n_{cl}$ ($\tilde{\chi}_{cl} \to 0$). At this point, the mode is no longer confined to the core and leaks into the cladding.
    \end{itemize}
\end{answer}

\begin{question}
    What is the cut-off frequency?
\end{question}

\begin{answer}
    The cut-off frequency is the lowest frequency at which a mode can propagate in a fiber.
    The condition is:
    $$\beta = k_{cl} \qquad \Rightarrow \qquad \begin{cases}
            \chi_{co} = \sqrt{k^2_{co} -\beta^2} = \dots = \frac{2\pi}{\lambda_0}NA \\
            \tilde{\chi}_{cl} \to 0
        \end{cases}$$

    Knowing the limit $\lim\limits_{u \to 0}\frac{uk_{n+1}(u)}{k_{n}(u)} = 2n$:
    $$\omega \text{ cut-off frequency } \qquad :\qquad \frac{\omega a}{c_0}(NA)\frac{J_{n+1}\left(\frac{\omega a}{c_0}(NA)\right)}{J_{n}\left(\frac{\omega a}{c_0}(NA)\right)} = 2n$$
    Which are the first solutions?
    $$LP_{0,1} =\begin{cases}
            n=0 \\
            \omega_c = 0
        \end{cases}\qquad \qquad LP_{1,1} = \begin{cases}
            n=1 \\
            \omega_{1,1}\approxeq\frac{2.405 c_0}{a(NA)}
        \end{cases}$$
    With Cut-off frequency we block propagation of other modes except $LP_{0,1}$.
\end{answer}

\begin{question}
    How can you verify the cutoff experimentally?
\end{question}

\begin{answer}
    The cutoff wavelength is experimentally verified using the **bend-loss method**.

    The principle of this method is that the modes closer to the cut-off are more sensitive to bend losses.

    By measuring the spectrum difference between a straight and a bent fiber, one can observe loss peaks. These peaks correspond to the wavelengths at which specific mode groups (like LP$_{11}$, LP$_{02}$, LP$_{21}$, etc.) are "stripped away" by the bend.

\end{answer}

\begin{question}
    Describe the properties and degeneracy of LP modes.
\end{question}

\begin{answer}
    Mode degeneracy occurs when multiple modes share the same propagation constant ($\beta$), which results from the fiber's cylindrical symmetry. The LP modes exhibit two types of degeneracy:

    \begin{enumerate}
        \item Polarization Degeneracy: Every $\text{LP}_{n,p}$ mode has two orthogonal polarization states that share the same $\beta$, resulting in a two-fold degeneracy.
        \item Polarization Degeneracy: Every $\text{LP}_{n,p}$ mode has two orthogonal polarization states that share the same $\beta$, resulting in a two-fold degeneracy.
    \end{enumerate}

    So at the end:
    \begin{itemize}
        \item $\text{LP}_{0,p}$ modes (where $n=0$) are two-fold degenerate (polarization only).
        \item $\text{LP}_{n,p}$ modes (where $n \geq 1$) are four-fold degenerate (polarization and azimuthal).
    \end{itemize}
\end{answer}

\chapter*{Attenuation}
\begin{question}
    What is attenuation?
\end{question}

\begin{answer}
    Attenuation, or gradual loss of signal strenght,stands as a critical limiting factor for fiber optic communication systems. In linear regime, losses can be described by a \textit{coefficient of attenuation}. Let $P(z)$ be the power flowing along the fiber. After an infinitersimally long section the power would be reduced by any amount proportional to it, according to:
    $$P(z+\Delta z) = P(z) - \alpha(z)\Delta zP(z)$$
    This leads to the fundamental relationship describing power loss:

    $$\frac{dP}{dz}=-\alpha(z)P(z)$$
    Solving this equation yields the general form of the Beer-Lambert Law:

    $$P(z)=P(0) e^{-\int_{0}^{z}\alpha(z')dz'}$$

    In the common case where the attenuation coefficient ($\alpha$) is constant (linear attenuation), the equation simplifies to its familiar exponential decay form:

    $$P(z)=P(0) e^{-\alpha z}$$

    Where:
    \begin{itemize}
        \item $P(z)$: Power at distance $z$ along the fiber.
        \item $P(0)$: Initial power at the fiber's start ($z=0$).
        \item $\alpha$: Attenuation coefficient (constant).
        \item $z$: Distance along the fiber.
    \end{itemize}
\end{answer}

\begin{question}

\end{question}

\begin{answer}

    Intrinsic Attenuation originates from the fundamental properties of pure silica glass and represents the theoretical minimum loss achievable. It includes:
    \begin{itemize}

        \item \textbf{Molecular Vibrations (IR Absorption)}: In the infrared region (around 1650 nm), light resonates with the natural vibrational states of silica molecules, leading to strong absorption.
        \item \textbf{Rayleigh Scattering}: The most important contribution to loss in the primary transmission windows, caused by microscopic, random fluctuations in the refractive index. Its magnitude has an inverse dependence on the fourth power of the wavelength ($\alpha_{\text{Rayleigh}} \propto 1/\lambda^4$).
    \end{itemize}

    Extrinsic Attenuation arises from impurities introduced during the manufacturing process and can be minimized through purification. It includes:

        \item \textbf{Molecular Hydrogen ($\text{H}_2$)}: Diffuses into the silica matrix, causing broad attenuation across all transmission, although this process is reversible.
        \item \textbf{Bending Losses}: Caused by physical curvature of the fiber, leading to the leak of light into the cladding. This is categorized into:

    \end{itemize}
\end{answer}

\begin{question}
    Discuss about bending losses.
\end{question}

\begin{answer}
    Bending losses are caused by physical curvature of the fiber, leading to the leak of light into the cladding. This is categorized into:
    \begin{itemize}
        \item \textbf{Macro-bending losses}: These occur when there is a discernible radius of curvature ($R$). A mode-based model identifies a critical distance ($X_c$) from the fiber's axis where the mode is "locally" at cut-off, causing the light to be radiated. The attenuation coefficient ($\alpha_{\text{bending}}$) has a severe exponential dependence on the bending radius ($R$).
        \item \textbf{Micro-bending losses}: This attenuation is caused by a continuous series of small, random bends. The physical mechanism involves the coupling of power from guided modes into random modes.
    \end{itemize}

\begin{question}
    How we describe the trasmission of a signal through a fiber? Frequency and Time domain.
\end{question}

\begin{answer}
    In frequency domain, we have a signal:
    $$\vec{E}(x,y,z,\omega) = A(\omega)\sum_{n=1}^{M}c_n \vec{E}_n(x,y)\exp\left\{-jz\beta_n(\omega)\right\}$$
    Since we want to work with the baseband signal, we have to work in $\omega + \omega_0$:
    $$\vec{E}(x,y,z,\omega) = A(\omega)\sum_{n=1}^{M}c_n \vec{E}_n(x,y)\exp\left\{-jz\beta_n(\omega + \omega_0)\right\}$$

    How do we get back to time domain? We have to Taylor-expand $\beta$ around $\omega_0$:
    $$\beta_n(\omega+\omega_0) \approx \beta_n(\omega_0) + \frac{dB_n}{d\omega}\big|_{\omega_0}\omega = \beta_{n,0} + \beta_{0,1}\omega$$
\end{answer}

\chapter*{Chromatic Dispersion}
\begin{question}
    Describe chromatic dispersion and a method to measure it
\end{question}

\begin{answer}
    Chromatic dispersion (CD) is a phenomenon that causes signal distortion in optical fibers. It arises because the propagation constant ($\beta$) is a function of the signal's angular frequency ($\omega$).

    For signals with a non-negligible bandwidth, $\beta(\omega)$ can be approximated using a Taylor series around the carrier frequency ($\omega_0$):

    $$\beta(\omega) \approx \beta_0 + \omega\beta_1 + \frac{1}{2}\omega^2\beta_2$$
    Where:
    \begin{itemize}
        \item $\beta_0$: phase constant at the carrier frequency.
        \item $\beta_1$: The inverse Group velocity dispersion parameter.
        \item $\beta_2$: The chromatic dispersion coefficient.
    \end{itemize}

    The physical origin of chromatic dispersion is the frequency-dependent group velocity ($V_g$), meaning $V_g$ changes with frequency if $\beta_2 \neq 0$.

    CD is categorized into two regimes:
    \begin{itemize}
        \item Normal dispersion: $\beta_2 \geq 0$; group velocity decreases as frequency increases.
        \item Anomalous dispersion: $\beta_2 < 0$; group velocity increases as frequency increases.
    \end{itemize}
    CD is alternatively described by the D parameter (Dispersion):

    $$D = \frac{d}{d\lambda}\left(\frac{d\beta}{d\omega}\right) = \frac{d}{d\lambda}\left(\frac{1}{V_g}\right)$$

    The D parameter and $\beta_2$ are related by the equation: $D = -\frac{2\pi c_0}{\lambda^2}\beta_2$.\\
    Chromatic Dispersion (CD) can be measured by measuring the propagation time for different optical wavelengths. This is based on the relationship $D L = \frac{d\tau_g}{d\lambda}$, where $D$ is the dispersion, $L$ is the fiber length, and $\tau_g$ is the group delay.
\end{answer}

\begin{question}
    What is a method to compensate chromatic dispersion?
\end{question}

\begin{answer}
    A method to compensate chromatic dispersion (CD) involves connecting fibers with opposite chromatic dispersion coefficients ($\beta_2$ or D parameter).

    This technique is called fibers concatenation.
    The process is:
    \begin{itemize}
        \item If the first fiber (A) has normal dispersion ($\beta_{A,2} > 0$), the spectral components that travel faster in fiber A are slowed down in the second fiber (B), which must have anomalous dispersion ($\beta_{B,2} < 0$).
        \item Conversely, the components that were slowed down in the first fiber are accelerated in the second fiber, equalizing the different frequency travel times.
        \item The overall chromatic dispersion of the concatenated link is determined by the sum of the $\beta_2 L$ products of each fiber. Complete compensation occurs when the condition $\beta_{A,2} L_A + \beta_{B,2} L_B \approx 0$ is met.
    \end{itemize}
\end{answer}

\begin{question}
    What are chirps and CPA?
\end{question}

\begin{answer}
    A pulse is said to be chirped if its carrier frequency changes with time.
    Chirped Pulse Amplification (CPA) is a technique that manages high-energy pulses by using chirping to lower the peak power before amplification.\\
    Four steps:
    \begin{itemize}
        \item Oscillator: Generates a short pulse with low energy.
        \item Stretcher: Takes the short, low-energy pulse and turns it into a long pulse with low energy, creating a strong chirp.
        \item Amplifier: Amplifies the long, low-energy pulse to a long pulse with high energy. The strong chirp lowers the pulse peak power, which prevents damage to the optical amplifier when amplifying to very high energy levels.
        \item Compressor: Takes the long, high-energy pulse and uses the reverse chirp to compress the pulse back to an ultra-short duration with high peak power.
    \end{itemize}
\end{answer}

\chapter*{Polarization Mode Dispersion}
\begin{question}
    Describe origins, effects and main parameters of PMD.
\end{question}

\begin{answer}
    Polarization Mode Dispersion (PMD) is a signal distortion phenomenon caused by the fact that standard optical fibers are not perfectly cylindrically symmetric. In perfectly symmetric fibers, the two orthogonal polarization states of the fundamental mode would have equal propagation constants.
    PMD is caused by factors that introduce asymmetries or stresses into the fiber, breaking the cylindrical symmetry.
    The main effect of PMD is Differential Group Delay (DGD).
    \begin{itemize}
        \item The fiber supports two orthogonal polarization states, called the Principal States of Polarization (PSP), which propagate without distortion.
        \item These two states travel at different group velocities, resulting in two replicas of the input signal arriving at different times.
        \item These two states travel at different group velocities, resulting in two replicas of the input signal arriving at different times.
        \item DGD is a random variable and is proportional to the square root of the fiber length (DGD $\propto \sqrt{L}$).
    \end{itemize}
    PMD is also characterized by the PMD matrix (Q), which summarizes the effect of polarization on propagation.

    \begin{itemize}
        \item The eigenvalues ($\tau_k$) of the PMD matrix correspond to the differential group delays.
        \item The eigenvectors ($\hat{v}_k$) of the PMD matrix correspond to the PSPs.
    \end{itemize}
    PMD is also represented by the PMD vector ($\overline{\Omega}$) on the Poincaré sphere.
\end{answer}

\begin{question}
    What is Polarization Mantaining Fiber (PMF)?
\end{question}

\begin{answer}
    A Polarization Maintaining Fiber (PMF) is a special type of optical fiber where the cylindrical symmetry is deliberately and uniformly broken along the fiber's length.

    This deliberate asymmetry allows for the definition of two distinct eigenstates, or principal states of polarization (PSP), which can propagate through the fiber unchanged.

    In this type of fiber, the magnitude of the PMD vector ($\overline{\Omega}$) is proportional to the fiber length ($L$) ($\overline{\Omega} \propto L$), unlike standard fibers where DGD is proportional to $\sqrt{L}$.
\end{answer}

\begin{question}

    We can analyze three types of interactions:
    \begin{itemize}
        \item Stimulated absorption: atom may absorb a photon and jumps to a higher energy state. The population changes according to:

                  \end{cases}$$
        \item Spontaneous emission: electrons from the top layer decay generating a photon of energy $h\nu = E_2 - E_1$.\\
              It is the source of noise in amplification.
              $$\frac{dN_2}{dt} = -\frac{1}{\tau_{sp}}N_2$$
        \item Stimulated emission: basically electrons decay generating duplicating the entering photons. In general $\sigma_{1,2} \neq \sigma_{2,1}$
              $$\frac{dN_2}{dt} = -\sigma_{2,1}\Phi N_2 \qquad \text{with }\begin{cases}
                      \sigma_{2,1} = \text{cross section}                         \\
                      \Phi = \text{flux: }\left[(\#\text{photons} / s)/m^2\right] \\
                      N_2 = \text{population of top layer}
                  \end{cases}$$


    \end{itemize}
\end{answer}

\begin{question}
    Explain the three-level system.
\end{question}

\begin{answer}
    Assume we have 3 levels, with many electrons that can move between the levels (given by doping material).
    \begin{itemize}
        \item From L3 (top) $\to$ L2 (middle): L3 instable, we have a quick decay to L2
        \item L2: Metastable, we need a sufficiently large $\tau_{sp}$ to store the electrons there
        \item Pump-photon: $h\nu = E_3 - E_1$, they promote electrons from L1 to L3
    \end{itemize}

\begin{question}
    Explain what are the rate equations.
\end{question}

\begin{answer}
    rate equations are needed to analyze the performance of the EDFA.\\
    What does it mean `to pump fast enough'? Suppose we have an infinitesimal cylindre of fiber:
    \begin{itemize}
        \item surface $s$
        \item length $\Delta z$
        \item in-flux $\Phi(z,t)$
        \item out-flux $\Phi(z+\Delta z, t + \Delta t (=\frac{\Delta z}{c}))$
    \end{itemize}
    So a first rate equation is given by \textbf{known in-flux} + \textbf{contribute of stimulated emissions} - \textbf{contribute of stimulated absorption} - \textbf{expected out-flux} = 0
    $$s\Delta t\left(\phi_s\left(z,t\right) - \phi_s\left(z + \Delta z, t + \frac{\Delta z}{c}\right)\right) + \left(s\Delta z\right)\Delta t\left[\sigma_s \phi_s\left(z,t\right)\left(N_2\left(z,t\right) - N_1\left(z,t\right)\right)\right] = 0$$
    At the end, after Taylor expansions, and knowing that $I_\square = h\nu_\square\phi_\square$:
    $$\text{Pump} : \frac{\partial I_p}{\partial z} + \frac{1}{c}\frac{\partial I_p}{\partial t} =- \sigma N_1I_p$$
    $$\frac{\partial I_s}{\partial z} + \frac{1}{c}\frac{\partial I_s}{\partial t} = \sigma(N_2 - N_1)I_s$$
    \textbf{Balance of population density}, we need to consider also spontaneous emissions. Since $N_{tot} = N_1 + N_2$:
    $$\frac{\partial N_2}{\partial t} = \frac{\sigma_p I_p}{h\nu_p}N_1 - \frac{\sigma_s I_s}{h \nu_s}(N_2 - N_1) - \frac{1}{\tau_{sp}}N_p$$

\end{answer}

\begin{question}
    Explain what are the rate equations for 3-level system.
\end{question}

\begin{answer}
    The rate equations for a 3-level system are used to analyze the performance of an Erbium Doped Fiber Amplifier (EDFA).

    When neglecting the time dependence, the rate equations for the signal intensity ($I_S$) and pump intensity ($I_P$) in the stationary regime are:

    Signal Intensity: $\frac{\partial I_S}{\partial z} = \sigma_S (N_2 - N_1)I_S$
    Pump Intensity: $\frac{\partial I_P}{\partial z} = -\sigma_p N_1 I_P$

    The population densities of the two levels, $N_1$ and $N_2$, are related to the total population density ($N_{tot}$) by the constraint $N_1(z, t) + N_2(z, t) = N_{tot}(z)$.

    In the stationary regime, the population density equation is set to zero ($\frac{\partial N_2}{\partial t} = 0$) and can be written as:

    $$0 = \frac{\sigma_p I_p}{h\nu_p}N_1 - \frac{\sigma_S I_S}{h\nu_S}(N_2 - N_1) - \frac{1}{\tau_{Sp}}N_2$$

    When these expressions for $N_1$ and $N_2$ are substituted back into the intensity equations, the rate equations in the stationary regime can be derived as:

    $$\frac{dI_p}{dz} = -\sigma_p N_{tot}I_p \frac{1 + \frac{I_S}{I_{S,Sat}}}{1 + \frac{I_p}{I_{p,Sat}} + 2 \frac{I_S}{I_{S,Sat}}}$$

    $$\frac{dI_S}{dz} = \sigma_S N_{tot}I_S \frac{\frac{I_p}{I_{p,Sat}} - 1}{1 + \frac{I_p}{I_{p,Sat}} + 2 \frac{I_S}{I_{S,Sat}}}$$
\end{answer}

\begin{question}
    What are the main Erbium parameters to check the performance of an EDFA?
\end{question}

\begin{answer}
    The main Erbium parameters related to the performance of an Erbium Doped Fiber Amplifier (EDFA) are:
    \begin{itemize}
        \item Gain ($G$): This is the ratio of the output signal power ($P_{out}(\nu_s)$) to the input signal power ($P_{in}(\nu_s)$).
        \item Power Conversion Efficiency ($PCE$): This is the ratio of the difference between the output and input signal power to the input pump power. The maximum theoretical value for PCE is $\frac{\nu_S}{\nu_P} = \frac{\lambda_P}{\lambda_S}$.
        \item Saturation Input Power ($P_{in}^{sat}$): This is the input signal power at which the EDFA gain decreases by 3 dB with respect to its maximum gain ($G_{max}$).
        \item Saturation Output Power ($P_{out}^{sat}$): This is the output signal power at which the EDFA gain decreases by 3 dB with respect to its maximum gain.
        \item Bandwidth: shape due to $\omega_s$, kind of a mountain shape on a wavelength-over-gain chart. It is the range of frequencies over which the gain is at least $G_{max} - 3 dB$.
    \end{itemize}
\end{answer}

\begin{question}
    What is ASE (Amplified Spontaneous Emission)?
\end{question}

\begin{answer}
    Amplified Spontaneous Emission (ASE) is a source of noise in an Erbium Doped Fiber Amplifier (EDFA).

    The equation that describes the average number of photons at signal frequency ($\langle n_S \rangle$), including both the signal and ASE, is:

    $$\frac{d\langle n_S\rangle}{dz}=\sigma_S(N_2-N_1)\langle n_S\rangle+2\sigma_S N_2$$

    where:
    \begin{itemize}
        \item $\sigma_S(N_2-N_1)\langle n_S\rangle$ is the average number of photons generated, per unit length, by stimulated emission at the signal frequency, minus the average number of photons absorbed, per unit length, by stimulated absorption at the signal frequency.
        \item $2\sigma_S N_2$ is the forcing term that generates ASE. This term qualitatively corresponds to the average number of photons generated, per unit length, by the emission stimulated by 2 spontaneous photons. The factor of 2 stems from the fact that a single-mode fiber has 2 degenerate modes.
    \end{itemize}
\end{answer}

\begin{question}
    What is the noise figure of an EDFA?
\end{question}

\begin{answer}
    The noise figure ($\text{NF}$) of an Erbium Doped Fiber Amplifier (EDFA) is a parameter used to quantify the degradation of the Optical Signal-to-Noise Ratio ($\text{OSNR}$) caused by the amplifier.

    The noise figure is defined as the ratio of the input OSNR to the output OSNR:
    $$\text{NF} = \frac{\text{OSNR}_{\text{in}}}{\text{OSNR}_{\text{out}}} > 1 \text{}$$

    In decibels ($\text{dB}$), the noise figure is expressed as $\text{NF}_{\text{dB}} > 3 \text{ dB}$, because for an EDFA, the noise figure is $\text{NF} > 2$, which represents the minimum possible noise (quantum noise).

    For an amplifier with gain $G \gg 1$, the noise figure can be approximated as:
    $$\text{NF} \approx 2 \cdot n_{\text{sp}} \text{}$$
    where $n_{\text{sp}} = \frac{N_2}{N_2 - N_1} \approx 1$ is the spontaneous emission factor.

    Generally, a higher gain amplifier has a lower noise figure. However, when the amplifier operates in the saturation regime, the gain is lower, and the noise figure is higher.

\end{answer}

\begin{question}
    How can EDFA be used for?
\end{question}

\begin{answer}
    \begin{itemize}
        \item As a \textbf{line amplifier} along the link or \textbf{pre-amplifier} before the receiver. The low power input signal is strongly amplified with minimal degradation of the OSNR.
        \item As a \textbf{booster amplifier} at the transmitter to increase the launch power into the fiber. Both NF and OSNR are high, so there is margin for degradation.
    \end{itemize}
\end{answer}

\chapter*{Optical Fibers Production and Non Linear Effects}

\begin{question}
    Describe the production of optical fibers.
\end{question}

\begin{answer}
    The production of an optical fiber cable consists of three main phases:

    \subsection*{1. Preform Production}
    The preform is a silica cylinder that is a larger-scale reproduction of the desired refractive index profile. It is typically about 1 meter long and ranges from a few up to a few tens of centimeters in diameter. A single preform can produce up to several hundreds of kilometers of optical fiber.

    \subsubsection*{Preform Production Techniques}
    There are two main techniques for producing the preform:
    \begin{itemize}
        \item \textbf{Chemical Vapor Deposition (CVD):} Silica is obtained by synthesis from a vapor phase. This is the most widely used system and enables the production of extremely pure silica with tight control over dopant concentration. Examples of CVD techniques include Inside Vapour Phase Oxidation (IVPO) processes like Modified Chemical Vapour Deposition (MCVD) and Plasma Chemical Vapour Deposition (PCVD), and Outside Vapour Phase Oxidation (OVPO) processes like Outside Vapour Deposition (OVD) and Vapour Axial Deposition (VAD).
        \item \textbf{Sol-Gel (Wet Phase):} Silica is produced from a wet phase. Silica and other constituents are initially held in a gel matrix and then sintered into the final preform. This is effective for obtaining silica with high levels of dopants or strong heterogeneity, and it has a higher yield than CVD techniques, though the final silica quality is lower.
    \end{itemize}

    \subsection*{2. Fiber Drawing}
    The preform is heated to its viscous state and then drawn into a fiber. This occurs in a special drawing tower, which is several meters tall.
    \begin{itemize}
        \item The preform is hung on a support and slowly enters a furnace, where it is heated near the silica melting temperature (about 2000°C).
        \item As the preform softens, a drop of silica begins to fall due to gravity, and the drawing process starts.
        \item The drawn fiber is cooled by passing it through cooling tubes with flows of inert gasses (like helium).
        \item The fiber diameter is a critical parameter that is constantly monitored to maintain the design range. Fiber diameter ($d_f$) is controlled by the drawing speed ($v_f$) based on the mass conservation equation:
              \begin{equation}
                  D_p^{2}V_{p}=d_{f}^{2}v_{f}
              \end{equation}
              where $D_p$ is the preform diameter and $V_p$ is the preform entry speed.
        \item The fiber is then coated with an acrylic layer (coating) to protect it from scratches. Standard telecommunication fibers have two coating layers: a soft inner layer for mechanical isolation and easy removal, and a hard outer layer for mechanical protection and low friction. The final outer diameter for standard telecom fibers is $250 \mu m$.
        \item The drawing speed for telecom fibers can be as high as several tens of meters per second.
    \end{itemize}

    \subsection*{3. Cabling}
    The coated optical fiber, with a standard outer diameter of 0.250 mm, requires further mechanical protection before installation.
    \begin{itemize}
        \item The structure of the cable varies greatly depending on the final application.
        \item The bare fiber is often protected by a $900 \mu m$ buffer layer of Teflon.
        \item Fibers can be grouped into flat ribbons for multi-fiber cables.
        \item In multi-fiber cables, the fibers are housed in tubes or slots carved into a supporting central dielectric rod, which also helps contrast thermal expansion.
        \item Cables often include a layer of water-swellable tape that expands upon contact with water to prevent infiltration.
        \item Metallic armoring can be used for protection against accidental hits and rodents.
        \item Submarine cables are heavily armored to withstand installation, high seabed pressure, and accidental impacts from anchors.
    \end{itemize}
\end{answer}

\begin{question}
    Describe the origins of main effects of nonlinearity in optical fibers
\end{question}

\begin{answer}
    The origins and main effects of nonlinearity in optical fibers can be described by examining the medium's polarization response to a strong electric field, which is often modeled using a mechanical oscillator approach.

    \subsection*{1. Origin of Nonlinearity}

    In a medium, when the intensity of the electric field becomes large enough, the standard approximation of a linear medium is no longer valid, and nonlinear terms must be accounted for.

    The total medium polarization ($\overline{p}$) is generally a function of the electric field ($\overline{e}(t)$). For small nonlinearities, this expression can be expanded in a Taylor series, where the medium polarization is given by:

    \begin{equation}
        \overline{p}=\epsilon_{0}(\chi^{(1)}\overline{e}+\chi^{(2)}:\overline{e}\overline{e}+\chi^{(3)}:\overline{e}\overline{e}\overline{e}+\cdots)
    \end{equation}

    The terms $\chi^{(n)}$ are the susceptibilities of the medium, with $\chi^{(1)}$ being the linear susceptibility. The $\chi^{(2)}$ and $\chi^{(3)}$ terms represent the quadratic and cubic nonlinear responses, respectively.

    \subsection*{2. Main Nonlinear Effects}

    The presence of the second-order ($\chi^{(2)}$) and third-order ($\chi^{(3)}$) susceptibilities gives rise to different nonlinear effects:

    \subsubsection*{A. Quadratic Term ($\chi^{(2)}$)}
    \begin{itemize}
        \item \textbf{Presence:} This term is present only in crystalline media, as it is zero in centro-symmetric media like glasses (silica fiber).
        \item \textbf{Effects:} The main effects due to the quadratic term are:
              \begin{itemize}
                  \item Second Harmonic Generation
                  \item Sum and Difference of Frequencies
                  \item Parametric Amplification
              \end{itemize}
    \end{itemize}

    \subsubsection*{B. Cubic Term ($\chi^{(3)}$)}
    \begin{itemize}
        \item \textbf{Presence:} This term is present in all media, including silica glass, which is the material for optical fibers.
        \item \textbf{Effects:} The main effects due to the cubic term are:
              \begin{itemize}
                  \item Third Harmonic Generation (though often inefficiently generated in fibers due to phase mismatch).
                  \item Self-Phase Modulation (SPM).
                  \item Cross-Phase Modulation (XPM).
                  \item Parametric Amplification and Wavelength Conversion.
                  \item Raman Scattering.
              \end{itemize}
    \end{itemize}

    \subsection*{3. Detailed Effects of Cubic Nonlinearity in Fibers}

    The cubic nonlinearity is the most relevant in standard optical fibers (which are centro-symmetric and made of silica glass).

    \paragraph{Self-Phase Modulation (SPM):}
    \begin{itemize}
        \item \textbf{Origin:} The term in the polarization oscillating at the same input frequency ($\omega_0$). The effect is that the nonlinear term does not modify the amplitude of the wave but only its phase.
        \item \textbf{Description:} The instantaneous frequency of the pulse varies (the pulse is "chirped"), and this chirp grows linearly with propagation distance ($z$). This effect is described by the time- and power-dependent nonlinear phase:
              \begin{equation}
                  \Phi_{NL}(z,t)=-\gamma P(t)z
              \end{equation}
    \end{itemize}

    \paragraph{Cross-Phase Modulation (XPM):}
    \begin{itemize}
        \item \textbf{Origin:} When two or more waves at different frequencies interact, the powerful wave can induce a nonlinear refractive index variation that is experienced by the other waves.
        \item \textbf{Description:} The nonlinear refractive index variation at a given frequency depends on the power of the wave at that frequency (SPM) and is twice as intense due to the power of the other wave (XPM). This effect is particularly important in Wavelength Division Multiplexing (WDM) systems.
    \end{itemize}

    \paragraph{Modulation Instability (MI):}
    \begin{itemize}
        \item \textbf{Origin:} An instability of a continuous wave (CW) solution to the Nonlinear Schrödinger Equation (NLSE) when a small perturbation is introduced.
        \item \textbf{Description:} For anomalous dispersion ($\beta_2 < 0$) and perturbation frequencies ($\omega$) below a critical frequency ($\omega_c$), the perturbation grows exponentially along the fiber. This growth is due to a nonlinear gain $g(\omega)$. This effect amplifies small perturbations like quantum noise, making the wave unstable.
    \end{itemize}
\end{answer}
