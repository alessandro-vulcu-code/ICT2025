\section{Divisibility}
\begin{proposition}[Properties of Divisibility]
Let $a, b, c \in \mathbb{Z}$. The following properties hold:
\begin{enumerate}
    \item If $a \mid 1$, then $a = \pm 1$.
    \item If $a \mid b$ and $b \mid a$, and $a \neq 0, b \neq 0$, then $a = \pm b$.
    \item If $a \mid b$ and $b \mid c$, then $a \mid c$ (Transitivity).
\end{enumerate}
\end{proposition}
\begin{proof}
    \leavevmode
\begin{enumerate}
    \item If $a \mid 1$: By definition, $\exists c \in \mathbb{Z} : 1 = ac$. Since $a, c \in \mathbb{Z}$, the only possible integer solutions are $(a=1, c=1)$ or $(a=-1, c=-1)$. Therefore, $a = \pm 1$.
    \item If $a \mid b$ and $b \mid a$: By definition, $\exists c_1, c_2 \in \mathbb{Z} : b = ac_1$ and $a = bc_2$. Substituting the first equation into the second gives $a = (ac_1)c_2 = a(c_1c_2)$. Since $a \neq 0$, we can divide by $a$ to get $1 = c_1c_2$. As $c_1, c_2 \in \mathbb{Z}$, the only solutions are $(c_1=1, c_2=1)$ or $(c_1=-1, c_2=-1)$.
    \begin{itemize}
        \item If $c_1 = 1$, then $b = a \cdot 1 = a$.
        \item If $c_1 = -1$, then $b = a \cdot (-1) = -a$.
    \end{itemize}
    Thus, $a = \pm b$.
    \item If $a \mid b$ and $b \mid c$: By definition, $\exists c_1, c_2 \in \mathbb{Z} : b = ac_1$ and $c = bc_2$. Substituting the first equation into the second gives $c = (ac_1)c_2 = a(c_1c_2)$. Since $c_1c_2 \in \mathbb{Z}$, this satisfies the definition of divisibility, so $a \mid c$. \qedhere
\end{enumerate}
\end{proof}

\begin{proposition}[Euclidean Division]
Let $a \in \mathbb{N}$ and $b \in \mathbb{N}$ with $b > 0$. Then $\exists! q, r \in \mathbb{Z} : a = bq + r$ with $0 \leq r < b$.
\end{proposition}

\begin{proposition}[Binary Representation from Repeated Division]
Let $m \in \mathbb{N}$, with $m < 2^B$. The repeated division-by-2 process:
\begin{align*}
m &= 2q_0 + m_0 \\
q_0 &= 2q_1 + m_1 \\
q_1 &= 2q_2 + m_2 \\
\dots
\end{align*}
terminates with $q_{B-1} = 0$, and $m = m_0 + 2m_1 + \dots + 2^{B-1}m_{B-1}$.
\end{proposition}
\begin{proof}
The proof consists of two parts: demonstrating the polynomial form and proving that the process terminates.

\subsubsection*{Polynomial Form}
We can show the relationship by algebraic substitution.
\begin{align*}
m &= 2q_0 + m_0 \\
\end{align*}
Substitute $q_0 = 2q_1 + m_1$:
\begin{align*}
m &= 2(2q_1 + m_1) + m_0 = 2^2q_1 + 2m_1 + m_0 \\
\end{align*}
Substitute $q_1 = 2q_2 + m_2$:
\begin{align*}
m &= 2^2(2q_2 + m_2) + 2m_1 + m_0 = 2^3q_2 + 2^2m_2 + 2m_1 + m_0
\end{align*}
Continuing this process until the final step where $q_{B-2} = 2q_{B-1} + m_{B-1}$, we arrive at the final form: $m = m_0 + 2m_1 + 2^2m_2 + \dots + 2^{B-1}m_{B-1}$.

\subsubsection*{Termination}
The process is guaranteed to terminate because the sequence of quotients is strictly decreasing and bounded.
\begin{itemize}
    \item Since $m < 2^B$, the first division $m = 2q_0 + m_0$ implies $2q_0 \leq m < 2^B$, so $q_0 < 2^{B-1}$.
    \item Similarly, $q_0 < 2^{B-1}$ implies $2q_1 \leq q_0 < 2^{B-1}$, so $q_1 < 2^{B-2}$.
\end{itemize}
This pattern continues: $q_i < 2^{B-1-i}$.
For the final quotient $q_{B-1}$, we have $q_{B-1} < 2^{B-1-(B-1)} = 2^0 = 1$. Since $q_{B-1} \in \mathbb{N} \cup \{0\}$, the only possibility is $q_{B-1} = 0$. The process terminates when the quotient becomes zero.
\end{proof}

\begin{remark}[Upper Bound of a B-bit Number]
If $m$ has $B$ bits in its binary representation, i.e., $m = (m_{B-1}\dots m_0)_2$, then $m < 2^B$.
\end{remark}
\begin{proof}
The largest possible value for a $B$-bit number occurs when all bits are $1$. This value can be calculated by summing the geometric series:
$$ m \leq \sum_{i=0}^{B-1} 2^i = \frac{2^B - 1}{2 - 1} = 2^B - 1 $$
Therefore, $m < 2^B$.
\end{proof}

\begin{corollary}[Range of a B-bit Number]
A natural number $m \in \mathbb{N}$ has a binary representation with $B$ bits (where the most significant bit $m_{B-1} = 1$) $\iff 2^{B-1} \leq m < 2^B$.
\end{corollary}
