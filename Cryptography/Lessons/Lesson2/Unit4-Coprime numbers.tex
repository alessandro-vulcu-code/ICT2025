\section{Coprime numbers}
\begin{proposition}[Characterisation of Coprime Numbers]
    Integers $a, b \in \mathbb{Z}$ are relatively prime $\iff \exists u, v \in \mathbb{Z} : a u + b v = 1$.
    \end{proposition}
    \begin{proof}
        \leavevmode
    \begin{itemize}
        \item[($\Rightarrow$)] This is a direct consequence of the Extended Euclidean Algorithm. If $\gcd(a, b) = 1$, then the algorithm guarantees $\exists u, v \in \mathbb{Z} : a u + b v = 1$.
        \item[($\Leftarrow$)] Conversely, assume $\exists u, v \in \mathbb{Z} : a u + b v = 1$. Let $d = \gcd(a, b)$. Since $d \mid a$ and $d \mid b$, it must divide their linear combination $a u + b v$. Therefore, $d \mid 1$. The only $d \in \mathbb{N}$ so that $d \mid 1$ is $d = 1$. \qedhere
    \end{itemize}
    \end{proof}

    \begin{corollary}
    Let $p$ be a prime number and $m \in \mathbb{Z}$. If $p \nmid m$, then $\gcd(p, m) = 1$.
    \end{corollary}
    \begin{proof}
    The positive divisors of a prime number $p$ are only $\{1, p\}$. If $p \nmid m$, then $p$ is not a common divisor. Therefore, the only common divisor of $p$ and $m$ is $1$, which means $\gcd(p, m) = 1$. \qedhere
    \end{proof}

    \begin{proposition}[Prime Divisor Property]
    Let $p$ be a prime and $a, b \in \mathbb{Z}$. If $p \mid a b$, then $p \mid a$ or $p \mid b$.
    \end{proposition}
    \begin{proof}
    Assume $p \mid a b$. If $p \mid a$, the proposition holds. If $p \nmid a$, then by the preceding corollary, $\gcd(p, a) = 1$.
    By the characterization of coprime numbers, $\exists u, v \in \mathbb{Z} : p u + a v = 1$.
    Multiplying this equation by $b$ gives $p u b + a b v = b$.
    \par
    Since $p \mid p u b$ and we are given $p \mid a b$, which implies $p \mid a b v$.
    Since $p$ divides both terms on the left side, it must divide their sum. Therefore, $p \mid (p u b + a b v) \rightarrow p \mid b$. \qedhere
    \end{proof}

    \begin{theorem}[Fundamental Theorem of Arithmetic]
    Let $a \in \mathbb{N}$ so that $a \geq 2$. Then $a$ can be factored as a product of primes, $a = p_1^{e_1}p_2^{e_2} \cdots p_m^{e_m}$, where $p_i$ are distinct primes, and $\forall i \in \{1, \dots, m\}, e_i \geq 1$.
    The factorization into primes is unique modulo rearrangements of the order of the factors.
    \end{theorem}

    \begin{remark}
    The exponent $e_i = \text{mult}_{p_i}(a)$ is the multiplicity of $p_i$ in $a$.
    A generalized form of the prime factorization is given by:
    $$ a = \prod_{p \text{ prime}} p^{\text{mult}_p(a)} $$
    where $\text{mult}_p(a) := 0$ if $p$ is not a prime factor of $a$.
    \end{remark}
