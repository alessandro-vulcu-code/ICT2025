% preamble.tex

% --- ENCODING AND LANGUAGE ---
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage[english]{babel}

% --- GEOMETRY AND PAGE LAYOUT ---
\usepackage{geometry}
\geometry{a4paper, margin=1in}
\usepackage{parskip}  % Adds space between paragraphs (block style)
\usepackage{setspace}
%\onehalfspacing % Uncomment to enable 1.5 line spacing

% --- FONT PACKAGES AND CONFIGURATION ---
\usepackage{fontspec}    % For custom fonts (required for \newfontfamily)
\usepackage{amsmath}     % For advanced math environments
\usepackage{unicode-math} % Necessary for xelatex/lualatex (loads after amsmath)
% Note: unicode-math provides all symbols from amssymb and amsfonts, so they are not needed
\usepackage{bm}          % For bold math symbols (\bm command)

% Set math font immediately after loading unicode-math
\setmathfont{Libertinus Math}

% Set text fonts
\setmainfont{Fira Sans}[
    UprightFont = Fira Sans Light,
    BoldFont = Fira Sans Regular,
]
\setsansfont{Fira Sans}

% --- GRAPHICS AND FIGURES ---
\usepackage{graphicx}
\usepackage{caption} % For customizing captions
\usepackage{standalone} % For compiling parts of documents independently
\usepackage{tkz-euclide} % This loads TikZ
\usetikzlibrary{fit} % Needed for the fit option
\usetikzlibrary{shapes.geometric}
\usetikzlibrary{decorations.pathmorphing} % For snake decoration used in TikZ

% --- TABLES ---
\usepackage{booktabs} % For professional tables
\usepackage{tabularx}
\renewcommand{\arraystretch}{1.4} % Increases vertical spacing between table rows

% --- HYPERLINKS AND REFERENCES ---
\usepackage{hyperref}

% --- THEOREMS AND BOXES ---
\usepackage{amsthm}
\usepackage{thmtools}
\usepackage[skins]{tcolorbox} % For colored boxes

% --- LISTS ---
\usepackage{enumitem}

% --- UTILITIES AND COLORS ---
\usepackage{xcolor}       % For coloring text and elements
\usepackage{fontawesome5} % For icons

% --- CODE LISTINGS (MINTED) ---
\usepackage{minted}   % For colored C++ code (requires Pygments and -shell-escape)
\setmonofont{Fira Code} % Set Fira Code as the monospace font
\setminted{
    fontsize=\small, % Makes the font slightly smaller
}
\usepackage{etoolbox} % For hooks to modify environments
\BeforeBeginEnvironment{minted}{\begin{center}}
\AfterEndEnvironment{minted}{\end{center}}

% --- HEADER AND FOOTER CONFIGURATION ---
\usepackage{fancyhdr} % For custom headers and footers

% Configure 'plain' style (used by \maketitle and \tableofcontents)
% to be consistent with the footer
\fancypagestyle{plain}{
    \fancyhf{} % Clears header and footer
    \cfoot{\thepage} % Puts only the page number at the center footer
    \renewcommand{\headrulewidth}{0pt} % No rule on these pages
}

% Configure Header/Footer for other pages
\pagestyle{fancy}
\fancyhead{} % Clears the header
\fancyfoot{} % Clears the footer
\fancyhead[L]{\nouppercase{\rightmark}} % Section name on the left (or Chapter)
\fancyhead[R]{\nouppercase{\leftmark}}  % Chapter name on the right (or Section)
\fancyfoot[C]{\thepage} % Page number in the center
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0pt}

% --- CUSTOM COMMANDS ---
\newcommand{\inputlesson}[1]{\clearpage\input{#1}\clearpage} % Command to input a lesson
