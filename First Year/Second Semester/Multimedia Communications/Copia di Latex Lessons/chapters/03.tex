\section{Introduction: From Samples to Bits}

The encoding pipeline for multimedia signals:

\begin{enumerate}
\item \textbf{Spatial Prediction}: reduces variance using local correlation — $y(n) = x(n) - \hat{x}(n)$
\item \textbf{Quantization}: maps high-precision residuals to discrete indices $i(n)$
\item \textbf{Lossless Coding}: maps indices to a compact bitstream
\end{enumerate}

A valid code must be:
\begin{itemize}
\item \textbf{Uniquely Decodable}: one-to-one mapping from bitstream back to symbol sequence
\item \textbf{Compact}: smallest possible average codeword length
\end{itemize}
\begin{center}\includegraphics[width=0.600\linewidth]{Pasted_image_20260501174916.png}\end{center}
\subsection{Fixed-Length Coding (FLC)}

All codewords have the same length $L = \lceil \log_2 M \rceil$ where $M$ = alphabet size.

\begin{center}
\begin{tabular}{|l|l|l|l|}
\hline
\textbf{Index} & \textbf{4-bit Code} & \textbf{Index} & \textbf{4-bit Code} \\ \hline
-7 & 1001 & 1 & 0001 \\ \hline
-6 & 1010 & 2 & 0010 \\ \hline
... & ... & ... & ... \\ \hline
0 & 0000 & 7 & 0111 \\ \hline
\end{tabular}
\end{center}

\begin{itemize}
\item \textbf{Pros}: simple parsing (every $L$ bits = one symbol), robustness (1 bit error affects 1 symbol only)
\item \textbf{Con}: fixed rate $R = \log_2 M$, ignores that symbols are not equally probable
\end{itemize}

\textgreater{} Example: French text with $M = 26$ letters → FLC needs 5 bpS, no compression.

\subsection{Variable-Length Coding (VLC)}

Assign \textbf{shorter codewords to probable symbols}, longer codewords to rare symbols.

Key questions:
\begin{enumerate}
\item \textit{Theoretical bound}: how much gain? → \textbf{Source Entropy} $H(X)$
\item \textit{Practical implementation}: \textbf{Huffman coding} (optimal, but scalability issues) or \textbf{Arithmetic coding} (asymptotically optimal, scales easily)
\item \textit{Unknown statistics}: \textbf{Universal coding} (LZW)
\end{enumerate}

\subsection{Terminology}

\begin{itemize}
\item \textbf{Alphabet}: $\mathcal{X} = \{x_1, x_2, \ldots, x_M\}$ — set of symbols to encode
\item \textbf{Code}: $C : x_i \in \mathcal{X} \rightarrow c_i \in \{0, 1\}^*$ — maps symbols to finite-length bit strings
\item $\ell_i$: length of codeword $c_i$
\end{itemize}

\textbf{Types of decodability}:
\begin{itemize}
\item \textit{Instantaneous} (prefix code): no codeword is prefix of another → can decode symbol-by-symbol
\item \textit{Uniquely decodable}: every finite sequence decodes unambiguously (may require lookahead)
\end{itemize}

\begin{calloutbox}{green!55!black}{Code Examples}
\begin{center}
\begin{tabular}{|l|l|l|l|l|}
\hline
\textbf{Symbol} & \textbf{Code 1} & \textbf{Code 2} & \textbf{Code 3} & \textbf{Code 4} \\ \hline
A (p=1/2) & 0 & 0 & 0 & 0 \\ \hline
B (p=1/4) & 0 & 1 & 10 & 01 \\ \hline

\item Code 1: \textbf{not} uniquely decodable (A and B share codeword 0; 10 = CA = CB = D)
\item Code 2: \textbf{not} uniquely decodable (00 = AA or C; 11 = BB or D)
\item Code 3: \textbf{instantaneously decodable} (prefix condition satisfied)
\item Code 4: decodable with delay of 1 bit
\end{itemize}
\end{calloutbox}

\section{Principles of Information Theory}

\subsection{Information and Entropy}

\textbf{Self-information} of symbol $x_i$ with probability $p_i$:

\[
I(x_i) = -\log p_i
\]

Properties:
\begin{itemize}
\item $I(x_i) \geq 0$
\item $p_i = 1 \Rightarrow I = 0$ (certain event carries no information)
\item Independent symbols: $I(x_i, x_j) = I(x_i) + I(x_j)$
\end{itemize}

\begin{calloutbox}{red!70!black}{Source Entropy}
\[
H(X) = -\sum_i p_i \log p_i
\]

\textbf{Interpretation:} average uncertainty (information) of the random variable $X$. Measured in bits when $\log = \log_2$.
\end{calloutbox}

For a \textbf{binary RV} with alphabet $\{0,1\}$:
\[
H(X) = -p \log p - (1-p)\log(1-p)
\]
Maximum at $p = 0.5$ where $H = 1$ bit.
\begin{center}\includegraphics[width=0.600\linewidth]{Pasted_image_20260501175049.png}\end{center}
\subsection{Maximum Entropy Distribution}

\begin{calloutbox}{red!70!black}{Maximum Entropy Theorem}
The distribution maximizing $H(X)$ for an $M$-ary discrete RV subject to $\sum p_i = 1$ is the \textbf{uniform distribution}:
\[
p_i^* = \frac{1}{M} \quad \Rightarrow \quad H_{\max} = \log_2 M
\]

\textbf{Proof (Lagrange multipliers):}
\[
J(p) = -\sum_{i} p_i \log p_i + \lambda\left(\sum_i p_i - 1\right)
\]
\[
\frac{\partial J}{\partial p_i} = 0 \Rightarrow \log p_i^* = \lambda - \log e = \text{const} \Rightarrow p_i^* = \frac{1}{M}
\]
\end{calloutbox}

\subsection{Joint and Conditional Entropy}

\textbf{Joint entropy} of $(X, Y)$:
\[
H(X, Y) = -\sum_{i,j} p_{i,j} \log p_{i,j}
\]

\textbf{Conditional entropy} (average uncertainty about $X$ given $Y$):
\[
H(X|Y) = \sum_j p_j \, H(X|Y = y_j)
\]

\textbf{Chain rule}:
\[
H(X, Y) = H(Y) + H(X|Y) = H(X) + H(Y|X)
\]

\begin{itemize}
\item $H(X|Y) = H(X)$ if $X, Y$ independent
\item $H(X|Y) = 0$ if $X$ is deterministic function of $Y$
\end{itemize}

\subsection{Properties}

\begin{center}
\begin{tabular}{|l|l|l|}
\hline
\textbf{Property} & \textbf{Formula} & \textbf{Equality condition} \\ \hline
Non-negativity & $H(X) \geq 0$ & always \\ \hline
Chain rule & $H(X,Y) = H(Y) + H(X|Y)$ & — \\ \hline
Sub-additivity & $H(X,Y) \leq H(X) + H(Y)$ & $X, Y$ independent \\ \hline
Conditioning reduces entropy & $H(X|Y) \leq H(X)$ & $X, Y$ independent \\ \hline
Upper bound & $H(X) \leq \log_2 M$ & $X \sim \mathcal{U}$ (uniform) \\ \hline
\end{tabular}
\end{center}

\begin{calloutbox}{red!70!black}{Conditioning Reduces Entropy}
$H(X|Y) \leq H(X)$ — "information never hurts": knowing $Y$ can only decrease (or leave unchanged) uncertainty about $X$.
\end{calloutbox}

\section{Optimal Coding}

\subsection{McMillan's Theorem and Kraft's Inequality}

\begin{calloutbox}{red!70!black}{McMillan's Theorem}
\textbf{Decodable codes do not improve performance with respect to instantaneous codes.} We can restrict attention to instantaneous (prefix) codes without loss of generality.
\end{calloutbox}

\begin{calloutbox}{red!70!black}{Kraft's Inequality}
A prefix code with codeword lengths $\{\ell_1, \ldots, \ell_M\}$ exists \textbf{if and only if}:
\[
\sum_{i=1}^{M} 2^{-\ell_i} \leq 1
\]

\begin{itemize}
\item Necessary and sufficient for existence of a prefix code
\item If $\sum = 1$: code is \textbf{complete} (maximal — no wasted tree branches)
\end{itemize}
\end{calloutbox}

\subsection{Proof of Kraft's Inequality}

\begin{center}\includegraphics[width=0.600\linewidth]{Pasted_image_20260522113842.png}\end{center}

\textbf{Necessity}: Build a binary tree of depth $L_{\max} = \max(\ell_i)$. Each codeword $c_i$ of length $\ell_i$ "blocks" $2^{L_{\max} - \ell_i}$ leaves. By prefix property, blocked sets are disjoint:

\[
\sum_{i=1}^{N} 2^{L_{\max} - \ell_i} \leq 2^{L_{\max}} \quad \Rightarrow \quad \sum_{i=1}^{N} 2^{-\ell_i} \leq 1
\]

\textbf{Sufficiency} (Greedy Construction):
\begin{enumerate}
\item Sort lengths $\ell_1 \leq \ell_2 \leq \cdots \leq \ell_N$
\item At step $k$, pick an available node at depth $\ell_k$
\item Mark all descendants as forbidden
\end{enumerate}

At each step, consumed capacity $= \sum_{i<k} 2^{-\ell_i} < 1$ guarantees a node is available.

\begin{calloutbox}{green!55!black}{Prefix Code Construction}
Lengths $\{1, 2, 3, 3\}$: $2^{-1} + 2^{-2} + 2^{-3} + 2^{-3} = 1$ (complete code)

Result: $\{0,\ 10,\ 110,\ 111\}$

\subsection{Shannon's Source Coding Theorem}

\textbf{Optimal code} minimizes average length:

\[
\ell^* = \arg\min_\ell \sum_i p_i \ell_i \quad \text{subject to} \quad \sum_i 2^{-\ell_i} = 1
\]

Lagrangian solution (relaxing $\ell_i \in \mathbb{N}$):

\[
J(\ell) = \sum_i p_i \ell_i + \lambda\left(\sum_i 2^{-\ell_i} - 1\right)
\]
\[
\frac{\partial J}{\partial \ell_i} = 0 \Rightarrow \ell_i^* = -\log_2 p_i
\]
\[
\mathcal{L}^* = \sum_i p_i \ell_i^* = H(X)
\]

\begin{calloutbox}{red!70!black}{Shannon's Source Coding Theorem}
The minimum average length of any lossless code satisfies:
\[
\mathcal{L}^* \geq H(X)
\]
with equality \textbf{if and only if} all probabilities are dyadic: $\forall i, \exists k \in \mathbb{N}: p_i = 2^{-k}$.

\textbf{Consequence:} Optimal coding is called \textbf{entropy coding}. The bound $H(X)$ is the fundamental limit.
\end{calloutbox}

\textbf{Practical bound}: using $\ell_i = \lceil -\log_2 p_i \rceil$ (integer lengths) gives:

\[
H(X) \leq \mathcal{L}^* < H(X) + 1
\]

\begin{calloutbox}{green!55!black}{French Text Compression}
\begin{itemize}
\item Alphabet size: 26 letters
\item FLC: 5 bpS
\item Source entropy: $H = 3.999$ bpS
\item Any VLC: $\mathcal{L} \geq 3.999$ bpS, compression ratio $\leq 1.25$
\end{itemize}
\end{calloutbox}

\section{Huffman Coding}
\subsection{Algorithm}

Huffman coding achieves \textbf{optimal} performance for known probabilities.

\textbf{Steps}:
\begin{enumerate}
\item Create leaf nodes for each symbol, weighted by $p_i$
\item Repeat until one node remains:
\begin{itemize}
\item Select two nodes with lowest weights
\item Create internal node as parent with weight = sum of children
\end{itemize}
\item Assign \texttt{0} to left edges, \texttt{1} to right edges
\item Read each codeword from root to leaf
\end{enumerate}

\begin{center}
\begin{tabular}{|l|l|l|l|}
\hline
\textbf{Symbol} & \textbf{Prob} & \textbf{Codeword} & \textbf{Length} \\ \hline
A & 0.40 & 0 & 1 \\ \hline
B & 0.20 & 100 & 3 \\ \hline
C & 0.15 & 101 & 3 \\ \hline
D & 0.15 & 110 & 3 \\ \hline
E & 0.05 & 1110 & 4 \\ \hline
F & 0.05 & 1111 & 4 \\ \hline
\end{tabular}
\end{center}

\[
\mathcal{L} = 0.4 \cdot 1 + 0.2 \cdot 3 + 0.15 \cdot 3 + 0.15 \cdot 3 + 0.05 \cdot 4 + 0.05 \cdot 4 = 2.3 \text{ bpS}
\]
\[
H \approx 2.2464 \text{ bpS}
\]

For French text with Huffman: $\mathcal{L} = 4.041$ bpS vs $H = 3.999$ bpS, ratio = 1.238.
\end{calloutbox}

\subsection{Huffman Coding Example}

\begin{calloutbox}{green!55!black}{Huffman Coding on a Black-and-White Image}
\textbf{Problem:} encode a black-and-white image where white pixels are much more frequent than black pixels.

\[
P(\square) = 86.7\%, \qquad P(\blacksquare) = 13.3\%
\]

The source entropy is:

\[
H(X) = 0.586 \text{ bpp}
\]

This means the average information per pixel is below one bit. However, if Huffman coding is applied to single pixels, the alphabet has only two symbols, so both symbols need one-bit codewords:

\begin{center}
\begin{tabular}{|l|l|}
\hline
\textbf{Symbol} & \textbf{Codeword} \\ \hline
$\blacksquare$ & \texttt{0} \\ \hline
$\square$ & \texttt{1} \\ \hline
\end{tabular}
\end{center}

Therefore:

\[
\mathcal{L} = 1 \text{ bpp}
\]

Single-symbol Huffman coding is valid, but inefficient here:

\[
\mathcal{L} - H(X) = 1 - 0.586 = 0.414 \text{ bpp}
\]

\begin{center}\includegraphics[width=0.600\linewidth]{Pasted_image_20260522123825.png}\end{center}
Figure: black-and-white T-shaped image with one-symbol Huffman table and probabilities $P(\square)=86.7\%$, $P(\blacksquare)=13.3\%$.

\textbf{Connection with predictive quantization:} prediction errors are usually concentrated around a few likely values. Fixed-length coding ignores this non-uniformity, while Huffman coding assigns shorter codewords to frequent residuals. This lowers the actual rate for the same reconstruction quality, and the entropy of quantized prediction errors gives the theoretical lower bound.

\textbf{Block-coding idea:} instead of encoding each pixel separately, encode a block of $K$ symbols as one larger symbol:

\[
X^K = (X_1, X_2, \ldots, X_K)
\]

Entropy is sub-additive:

\[
H(X_1, X_2, \ldots, X_K) \leq \sum_{i=1}^{K} H(X_i)
\]

Equality holds only when components are independent. Huffman coding on blocks satisfies:

\[
\mathcal{L} < H(X^K) + 1
\]

Dividing by $K$ gives the average length per original symbol:

\[
\mathcal{L}_S < \frac{H(X^K)}{K} + \frac{1}{K}
\]

The $+1$ bit Huffman penalty is now spread over $K$ pixels, so the per-pixel overhead becomes $1/K$.

\textbf{Two-pixel blocks:} possible symbols are:

\begin{center}
\begin{tabular}{|l|l|}
\hline
\textbf{Block} & \textbf{Probability} \\ \hline
$BB$ & $p^2$ \\ \hline
$WB$ & $p(1-p) \approx p$ \\ \hline
$BW$ & $p(1-p) \approx p$ \\ \hline
$WW$ & $(1-p)^2 \approx 1 - 2p$ \\ \hline
\end{tabular}
\end{center}

Since white pixels dominate, $WW$ is most likely and receives the shortest code. A possible Huffman assignment is:

\begin{center}
\begin{tabular}{|l|l|}
\hline
\textbf{Block} & \textbf{Codeword} \\ \hline
$WW$ & \texttt{0} \\ \hline
$WB$ & \texttt{10} \\ \hline
$BW$ & \texttt{110} \\ \hline
$BB$ & \texttt{111} \\ \hline
\end{tabular}
\end{center}

This gives approximately:

\[
\mathcal{L} \approx 1 \text{ bit per block}, \qquad \mathcal{L}_S \approx \frac{1}{2} \text{ bpp}
\]
\begin{center}\includegraphics[width=0.600\linewidth]{Pasted_image_20260522124215.png}\end{center}
Figure: two-pixel block example with block probabilities and Huffman assignment.

For the T-shaped image, measured two-pixel block probabilities are:

\begin{center}
\begin{tabular}{|l|c|l|}
\hline
\textbf{Block} & \textbf{Probability} & \textbf{Codeword} \\ \hline
$\square\square$ & $80.0\%$ & \texttt{1} \\ \hline
$\square\blacksquare$ & $5.00\%$ & \texttt{010} \\ \hline
$\blacksquare\square$ & $5.00\%$ & \texttt{011} \\ \hline
$\blacksquare\blacksquare$ & $10.0\%$ & \texttt{00} \\ \hline
\end{tabular}
\end{center}

The block entropy is:

\[
H([X_1 X_2]) = 1.022
\]

Per pixel:

\[
\frac{H([X_1 X_2])}{2} = 0.511 \text{ bpp}
\]

Average Huffman length:

\[
\mathcal{L}
= 0.80 \cdot 1 + 0.05 \cdot 3 + 0.05 \cdot 3 + 0.10 \cdot 2
= 1.3 \text{ bits per block}
\]

Therefore:

\[
\mathcal{L}_S = \frac{1.3}{2} = 0.65 \text{ bpp}
\]

Two-pixel block coding improves the rate:

\[
1.00 \text{ bpp} \rightarrow 0.65 \text{ bpp}
\]

but still remains above the entropy limit:

\[
0.65 \text{ bpp} > 0.511 \text{ bpp}
\]

\textbf{Four-pixel blocks:} increasing the block size improves the match between Huffman code lengths and source probabilities.

\[
H([X_1 X_2 X_3 X_4]) = 1.533
\]

\[
\frac{H([X_1 X_2 X_3 X_4])}{4} = 0.383 \text{ bpp}
\]

The Huffman length is:

\[
\mathcal{L} = 1.733 \text{ bits per block}
\]

Per pixel:

\[
\mathcal{L}_S = \frac{1.733}{4} = 0.433 \text{ bpp}
\]

Four-pixel coding gets closer to entropy:

\[
0.433 \text{ bpp} > 0.383 \text{ bpp}
\]

\begin{center}\includegraphics[width=0.600\linewidth]{Pasted_image_20260522124323.png}\end{center}
Figure: four-pixel block example with 16 possible black-white patterns, entropy $1.533$ bits per block, and Huffman rate $0.433$ bpp.

\textbf{Takeaway:} Huffman coding is optimal for the chosen alphabet, but the alphabet choice matters. Single-pixel coding has coarse one-bit code lengths; block coding creates richer symbols and distributes the Huffman overhead across several pixels, so the practical rate moves closer to entropy.
\end{calloutbox}
\subsection{Limits of Huffman}

\begin{itemize}
\item Complexity \textbf{exponential} in $K$ (alphabet size $M^K$)
\item Joint probability estimation for large blocks is costly and unreliable
\item Cannot handle the case $H(X) < 1$ efficiently symbol by symbol
\end{itemize}

→ \textbf{Arithmetic coding} solves at least the complexity problem.

\section{Arithmetic Coding}

\subsection{Principle}

Arithmetic coding performs \textbf{block coding with linear complexity} by encoding entire sequences as a single interval in $[0, 1)$.

\textbf{Encoding procedure}: each sequence of $n$ symbols determines a sub-interval according to its probability. For each new symbol, the interval is refined using 2 multiplications and 2 sums. The final codeword is the interval center, represented with sufficient precision.

\begin{calloutbox}{green!55!black}{Arithmetic Coding: Input ACFD}
Probabilities: A=0.4, B=0.2, C=0.15, D=0.15, E=0.05, F=0.05

\begin{itemize}
\item Start: $[0, 1)$
\item A: $[0, 0.4)$, center $= 0.2$, precision $= p_1/2$
\item C: $[0.24, 0.30)$, center $= 0.275$, precision $= p_1 p_3/2$
\item F: $[0.297, 0.30)$, center $= 0.2985$
\item D: $[0.29925, 0.2997)$, center $= 0.299475$
\end{itemize}
\end{calloutbox}

\subsection{Average Length}

\[
L(n) = -\left\lceil\sum_{i=1}^n \log_2 p(x_i)\right\rceil + 1 < -\sum_{i=1}^n \log_2 p(x_i) + 2
\]

Taking expectation:
\[
\mathcal{L} < H(X) + \frac{2}{n} \xrightarrow{n\to\infty} H(X)
\]

Arithmetic coding is \textbf{asymptotically optimal}, with a penalty of only $+2$ bits per block (vs. Huffman's $+1$ per symbol).

\subsection{Adaptivity and Context-Based Coding}

\textbf{Adaptivity}: symbol statistics are learned during encoding by maintaining occurrence counts, updated at both encoder and decoder by an agreed rule.

\textbf{Context-based coding}: instead of estimating $P(X^K)$ directly (hard for large $K$), use a \textbf{context} — the $N_S$ previous symbols that most influence the current one.

\begin{itemize}
\item Number of possible contexts: $N_C = M^{N_S}$
\item Using $N_C$ contexts ≡ $N_C$ different arithmetic encoders switching between each other
\item Reaches entropy rate $\mathcal{H}$ without massive block sizes
\end{itemize}

\begin{calloutbox}{green!55!black}{Context Coding on B/W Image}
Using previous pixel as context:
\begin{itemize}
\item $P(\square|\square) = 94.1\%$, $P(\blacksquare|\square) = 5.9\%$
\item $P(\square|\blacksquare) = 33.3\%$, $P(\blacksquare|\blacksquare) = 66.7\%$
\item $H(X|\square) = 0.322$, $H(X|\blacksquare) = 0.918$
\item $H(X|Y) = 0.406$ bpp — better than block coding $K=4$ (0.383 limit, 0.433 practical)!
\end{itemize}
\end{calloutbox}

\textbf{Design rule}: context must capture the "significant past". Too large → sparse data, unreliable estimation. Too small → fails to capture dependencies.

\subsection{Arithmetic Coding Conclusions}

\textbf{Advantages}:
\begin{itemize}
\item Linear complexity encoding/decoding ($O(n)$ operations)
\item Block coding: exploits high-order dependencies, removes non-dyadic penalty
\item Context coding: simple model for high-order statistics
\item Adaptivity: handles non-stationary sources
\end{itemize}

\textbf{Disadvantages}:
\begin{itemize}
\item Implementation tricky (precision, carry propagation)
\item Needs initialization
\item Context selection is non-trivial
\item Adaptivity needs large training set for robust estimation
\end{itemize}

\section{Other Techniques}

\subsection{Exponential-Golomb Coding}

\textbf{Universal coding} — no initialization, no need to specify symbol range in advance.

\textbf{Unsigned integers} — encoder for $n \geq 0$:
\begin{enumerate}
\item $n = 0$: encode as \texttt{1}
\item $n \geq 1$: write $n+1$ in binary using $b = \lfloor\log_2(n+1)\rfloor + 1$ bits; prepend $b-1$ zeros
\end{enumerate}

\begin{center}
\begin{tabular}{|l|l|l|}
\hline
\textbf{$n$} & \textbf{$n+1$ (binary)} & \textbf{Codeword $c_U(n)$} \\ \hline
0 & 1 & \texttt{1} \\ \hline
1 & 10 & \texttt{010} \\ \hline
2 & 11 & \texttt{011} \\ \hline
3 & 100 & \texttt{00100} \\ \hline
7 & 1000 & \texttt{0001000} \\ \hline
\end{tabular}
\end{center}

\textbf{Signed integers} — map $n \in \mathbb{Z}$ to $m \in \mathbb{N}$:
\[
m(n) = \begin{cases} 2n - 1 & \text{if } n > 0 \\ -2n & \text{if } n \leq 0 \end{cases}
\]
Then encode $m$ with unsigned Exp-Golomb. Decoding inverts the mapping.

\textbf{Use cases}: metadata encoding, sparse residuals (e.g., prediction errors in DPCM).

\subsection{Dictionary-Based Coding (LZW)}

\textbf{Core idea}: give up pre-established probabilities; learn long-term dependencies during encoding via a \textbf{dynamic dictionary} of patterns.

\begin{calloutbox}{red!70!black}{Asymptotic Optimality Theorem}
For any stationary and ergodic source, dictionary-based coding is \textbf{asymptotically optimal}:
\[
L_n \xrightarrow{n\to\infty} \mathcal{H}(X)
\]
\end{calloutbox}

\textbf{Key characteristics}:
\begin{itemize}
\item \textbf{Dynamic}: dictionary built on-the-fly — no prior statistics required
\item \textbf{Universal}: works for any source
\item \textbf{Adaptive}: tracks non-stationary signals
\item \textbf{Practical}: basis of ZIP, GZIP, PNG (Deflate)
\end{itemize}

\textbf{LZW Encoding}:
\begin{enumerate}
\item Initialize dictionary with single-symbol entries
\item Read symbols; find longest prefix $W$ already in dictionary
\item If $W + K$ (current prefix + next symbol) is new: output index of $W$, add $W+K$ to dictionary
\item Continue with $K$ as new prefix
\end{enumerate}

\textbf{LZW Decoding}: mirrors encoding; decoder builds the same dictionary in sync.

The output values are dictionary indices, not final binary codewords. The compression benefit appears when a long pattern is represented by one index instead of by all its raw bits.

\textbf{Decoding rule:} read each index, output the associated pattern, and add the same new patterns in the same order. The decoder does not receive the dictionary; it reconstructs it because encoder and decoder follow the same deterministic update rule.

A shorter coded prefix is decoded as:
\begin{center}
\texttt{0 2 1 2 2 0}
\end{center}

During decoding, some just-created entries are written with temporary symbols such as \texttt{x}, \texttt{y}, \texttt{z}, and \texttt{w}. These placeholders mean: "this bit will be known when the corresponding dictionary entry is resolved." This is the classic LZW self-reference case.

The decoder evolution is:
\begin{center}
\begin{tabular}{|l|l|l|}
\hline
\textbf{Input prefix read} & \textbf{New dictionary entry} & \textbf{Decoded output form} \\ \hline
\texttt{0} & \texttt{2 = 0x} & \texttt{0x} \\ \hline
\texttt{0 2} & \texttt{3 = 0xy}, then \texttt{3 = 00y} & \texttt{000y} \\ \hline
\texttt{0 2 1} & \texttt{4 = 1z} & \texttt{0001z} \\ \hline
\texttt{0 2 1 2} & \texttt{5 = 00w} & \texttt{000100w} \\ \hline
\end{tabular}
\end{center}

\end{center}

\end{calloutbox}

\subsection{Standards: JBIG-1 and JBIG-2}

\textbf{JBIG-1} (ISO/IEC 11544, ITU-T T.82):
\begin{itemize}
\item Context-based arithmetic coding for bi-level images
\item Template of 10 pixels with 1 variable position
\item Progressive coding (resolution scalability)
\end{itemize}

\textbf{JBIG-2} (ISO/IEC 14492, ITU-T T.88):
\begin{itemize}
\item Image segmented into: text, halftones, other
\item Text: symbol dictionary coding
\item Halftone: dictionary of halftone patterns
\item Other: context-based arithmetic coding
\item Used in PDF (version 1.4+)
\end{itemize}

\subsection{Image Compression: JPEG-LS and PNG}

\textbf{JPEG-LS} (based on LOCO-I algorithm) — low complexity lossless for natural images:

\textit{MED Predictor} (Median Edge Detector) using causal neighborhood $(A, B, C)$ for pixel $X$:

\[
\hat{x} = \begin{cases} \min(A, B) & \text{if } C \geq \max(A, B) \\ \max(A, B) & \text{if } C \leq \min(A, B) \\ A + B - C & \text{otherwise} \end{cases}
\]

\begin{center}
\begin{tabular}{|l|l|}
\hline
\textbf{C} & \textbf{B} \\ \hline
A & X \\ \hline
\end{tabular}
\end{center}

Designed to handle sharp edges without overshoot.

\textbf{Residual coding}: context modeling via local gradients → \textbf{Golomb-Rice coding} for residuals → run-length encoding for uniform regions.

\section{Neural Lossless Coding}

\subsection{Density Estimation Perspective}

Neural Lossless Coding (NLC) uses a \textbf{Neural Network (NN) as a probability distribution estimator} $Q$ for an arithmetic coder.

The NN minimizes the \textbf{Cross-Entropy} between true distribution $P$ and model $Q$:

\[
H(P, Q) = H(P) + D_{KL}(P \parallel Q)
\]

\begin{itemize}
\item $H(P)$: source entropy (physical limit, irreducible)
\item $D_{KL}(P \parallel Q)$: penalty for model inaccuracy (redundancy to minimize)
\end{itemize}

\textbf{Advantage over classical context coding}:
\begin{itemize}
\item Classical: state space explodes exponentially ($M^{N_S}$ entries for $N_S$-pixel context at $M$ levels)
\item Neural: maps contexts to continuous vector space, generalizes to unseen patterns
\item Transformers/attention: capture \textbf{global dependencies} across the entire image
\end{itemize}

\subsection{Autoregressive Models}

\textbf{Mathematical foundation} (Chain Rule of probability):

\[
P(x_1, \ldots, x_n) = \prod_{i=1}^n P(x_i \mid x_1, \ldots, x_{i-1})
\]

The NN estimates $P(x_i \mid x_1, \ldots, x_{i-1})$ for each pixel given all previous (causal) pixels. Bit-rate approaches entropy rate $\mathcal{H}$.

\textbf{Trade-off}: high compression, but $N$ forward passes for $N$-pixel image → very high decoding complexity.

\subsection{Latent Variable Models}

Based on \textbf{Variational Inference}: image $x$ mapped to lower-dimensional latent $z$ (non-linear PCA).

\textbf{Two-stage coding}:
\begin{enumerate}
\item Encode latent features $z$ (summary)
\item Encode $x$ given $z$ (residuals/likelihood)
\end{enumerate}

\textbf{Bits-back principle}: bits used to specify $z$ can be "recovered" from randomness of previous message (in a sequence of images).

Performance bounded by \textbf{ELBO} (Evidence Lower Bound).

\subsection{Neural Predictive Coding}

Evolution of classical DPCM: replace linear predictor with a non-linear NN.

\[
\hat{x}_i = f_{NN}(\text{Local Context})
\]

Trained to minimize \textbf{cross-entropy of residual distribution} (not MSE).

\textbf{MLP architecture} for the house experiment:
\begin{itemize}
\item Input: causal neighborhood $S = \{A, B, C\}$
\item Output: $\hat{Y} = f_{MLP}(S, W)$
\item Final code: prediction rounded to integer residuals
\end{itemize}

\textbf{Results on house.pgm}:
\begin{itemize}
\item 2D linear predictor: residual entropy = 2.830 bpp
\item Neural predictor (MLP): residual entropy = \textbf{2.704 bpp}
\end{itemize}

\textit{The NN handles complex textures and non-linear transitions that the deterministic condition ($|C-B| < |C-A|$) misses.}

\textbf{Total rate accounting for model transmission cost}:

\[
R_{\text{total}} = R_{\text{residuals}} + \frac{L_{\text{model}}}{N}
\]

\begin{calloutbox}{green!55!black}{Model Cost}
Small MLP [10, 5] with 3 inputs ≈ 100 weights. Using 32-bit floats:
\[
L_{\text{model}} = 3200 \text{ bits}
\]
For $512 \times 512$ image:
\[
R = 2.704 + \frac{3200}{512^2} \approx 2.716 \text{ bpp}
\]

NLC more efficient for large datasets (weights shared) or high-resolution images.
\end{calloutbox}

\section{Conclusions}

\subsection{Comparison of All Lossless Coding Techniques}

\begin{center}
\begin{tabular}{|l|l|l|l|l|}
\hline
\textbf{Method} & \textbf{Efficiency} & \textbf{Complexity} & \textbf{Latency} & \textbf{Best Use Case} \\ \hline
Neural (SOTA) & Highest & Ultra High & High & SOTA image/video \\ \hline
Arithmetic & High (→ $H$) & High (CPU) & Medium & Bi-level images, CABAC in H.264/H.265 \\ \hline
Dictionary (LZW) & Universal & Moderate & Low/Med & Repeating patterns, general files \\ \hline
Huffman & Good ($H \leq \mathcal{L} < H+1$) & Low/Med & Very Low & JPEG, general purpose \\ \hline
Exp-Golomb & Approximate & Very Low & Negligible & Metadata, sparse residuals \\ \hline
\end{tabular}
\end{center}

\textbf{The 1-bit penalty}:
\begin{itemize}
\item Huffman fails when $H(X) < 1$ bpS (one symbol encodes as 1 bit minimum)
\item Arithmetic coding efficient even below 1 bpS
\end{itemize}

\subsection{Lossless Design Matrix}

\begin{center}
\begin{tabular}{|l|l|l|}
\hline
\textbf{Source characteristic} & \textbf{Recommended method} & \textbf{Reason} \\ \hline
Memoryless & Huffman / Arithmetic & Respects $H(X)$ bound \\ \hline
Stationary with memory & Context-Adaptive Arithmetic & Optimal for Markov statistics \\ \hline
Locally stationary & Adaptive Arithmetic & Tracks changing statistics \\ \hline
Highly complex dependencies & Neural Models & Captures non-linear, long-range dependencies \\ \hline
\end{tabular}
\end{center}

\textbf{All methods aim for} $\mathcal{L} \approx \mathcal{H}$. Choice depends on complexity budget, latency requirements, and model cost (transmitting NN weights vs. compression gain).

\textgreater{} \textbf{Final rule of thumb:} Use Huffman/EG for speed, Arithmetic/Context for high compression, Dictionary for universal files, and Neural Models for ultimate SOTA performance.
