\subsection{Light Perception}

\begin{itemize}
\item Perceived intensity: \textbf{logarithmic function} of physical intensity
\item Physical dynamic range: $\approx 10^{10}$ (100 dB)
\item HVS cannot operate over the full range simultaneously → adapts via \textbf{global lighting changes} (reduced dynamic range at any given moment)
\item Perceived brightness is not a simple function of luminance — context matters
\end{itemize}

\subsection{Contrast Sensitivity Function (CSF)}

\begin{calloutbox}{red!70!black}{Contrast Sensitivity Function (CSF)}
The CSF measures HVS ability to detect luminance differences across \textbf{spatial frequencies}.

\begin{itemize}
\item \textbf{Contrast}: difference in luminance between lightest and darkest parts of a pattern (\%)
\item \textbf{Spatial frequency}: measured in \textbf{cycles per degree} of visual angle
\item \textbf{Sensitivity} = $1 / \text{(minimum detectable contrast)}$
\end{itemize}

CSF \textbf{peaks at mid-range frequencies} (≈ 2–5 cycles/degree) and declines at both low and high frequencies.

\textbf{Implication for compression:} high-frequency errors are less visible → can apply coarser quantization to high frequencies.
\end{calloutbox}
\begin{center}\includegraphics[width=0.806\linewidth]{Pasted_image_20260501163356.png}\end{center}
\subsection{Color Perception}

\begin{itemize}
\item Visible light: \textbf{400–700 nm}
\item Cone distribution by color sensitivity:
\begin{itemize}
\item 65\% sensitive to \textbf{red} (\textasciitilde{}575 nm)
\item 33\% sensitive to \textbf{green} (\textasciitilde{}535 nm)
\item 2\% sensitive to \textbf{blue} (\textasciitilde{}445 nm) — but highly sensitive
\end{itemize}
\end{itemize}
\begin{center}\includegraphics[width=0.600\linewidth]{Pasted_image_20260501163417.png}\end{center}

\begin{itemize}
\item \textbf{Tristimulus}: color sensation corresponds to the 3D response vector of the three cone types
\item Any color = weighted combination of three primary colors
\end{itemize}

\subsection{Color Spaces}

\textbf{RGB} — unit cube representation:
\begin{itemize}
\item $(0,0,0)$: Black; $(1,1,1)$: White; main diagonal: Grayscale
\item Primaries at $(1,0,0)$, $(0,1,0)$, $(0,0,1)$
\item Secondaries: Cyan $(0,1,1)$, Magenta $(1,0,1)$, Yellow $(1,1,0)$
\end{itemize}
\begin{center}\includegraphics[width=0.600\linewidth]{Pasted_image_20260501163800.png}\end{center}
\textbf{HSV} — derived by balancing RGB cube on black corner, rotating so black-white diagonal is vertical, projecting:
\begin{itemize}
\item \textbf{Hue} (H): angle 0°–360° around vertical axis → color family
\item \textbf{Saturation} (S): distance from central axis → color purity
\item \textbf{Value} (V): vertical position 0–1 → brightness
\end{itemize}

\textbf{YCbCr} (≈ digital YUV):
\begin{itemize}
\item \textbf{Y}: luminance (same as HSV Value)
\item \textbf{Cb, Cr}: chrominance (Cartesian representation vs. polar S/H in HSV)
\item $Cb \approx B - Y$, $Cr \approx R - Y$
\end{itemize}

\subsection{RGB to YCbCr Conversion}

\begin{calloutbox}{red!70!black}{ITU-R BT.601 Color Conversion}
\[
\begin{bmatrix} Y \\ Cb \\ Cr \end{bmatrix} = \begin{bmatrix} 0.299 & 0.587 & 0.114 \\ -0.1687 & -0.3313 & 0.5 \\ 0.5 & -0.4187 & -0.0813 \end{bmatrix} \begin{bmatrix} R \\ G \\ B \end{bmatrix} + \begin{bmatrix} 0 \\ 128 \\ 128 \end{bmatrix}
\]

\begin{itemize}
\item \textbf{Y weights} reflect HVS sensitivity: green weighted most (0.587), blue least (0.114)
\item \textbf{Cb, Cr offset} (+128) centers the 8-bit range $[0,255]$
\item \textbf{Invertible}: $\mathbf{p}_{RGB} = M^{-1}(\mathbf{p}_{YCbCr} - \mathbf{o})$ — perfect reconstruction
\end{itemize}
\end{calloutbox}

\subsection{Beyond HVS: Machine-Centric Multimedia}

A growing fraction of multimedia content is consumed by \textbf{algorithms} (surveillance, IoT, industrial monitoring), not humans.

\begin{center}
\begin{tabular}{|l|l|l|}
\hline
\textbf{Aspect} & \textbf{Human Visual System} & \textbf{Machine Models} \\ \hline
Priority & Semantic regions (faces), CSF-limited & High-frequency details, local textures \\ \hline
Goal & Minimize perceived distortion & Maximize downstream task accuracy \\ \hline
Compression strategy & Psychovisually lossless & Discard "semantically irrelevant" content \\ \hline
\end{tabular}
\end{center}

\textbf{Task-Oriented Communications}: if the end-user is a machine, extreme compression ratios are possible by discarding visually important but semantically irrelevant information.

\section{Sound Perception}

\subsection{Pure Tone Perception}

A pure tone $x(t) = a\sin(2\pi f_1 t)$ with power $\sigma^2 = a^2/2$ excites \textbf{multiple nerve fibers} simultaneously (power spreading).

\textbf{Filterbank model}: $M$ filters, each corresponding to one nerve fiber $k$:
\begin{itemize}
\item Frequency response: $H_k(f) = A_k(f)\exp^{j\phi_k(f)}$
\item Response to tone at $f_1$: $y_k(t) = a A_k(f_1)\sin[2\pi f_1 t + \phi_k(f_1)]$
\item \textbf{Spreading function}: $S_E(k) = A_k^2(f_1)$ — power distribution across fibers
\end{itemize}

\subsection{Hearing Threshold}
\begin{center}\includegraphics[width=0.600\linewidth]{Pasted_image_20260501163839.png}\end{center}
\begin{itemize}
\item \textbf{Absolute hearing threshold} $S_a(f)$: minimum power for a tone at frequency $f$ to be audible
\item Minimum of $S_a(f)$ at 1–4 kHz → HVS most sensitive in speech frequency range
\item Compression implication: components below $S_a(f)$ can be discarded
\end{itemize}

\subsection{Critical Band}

\begin{calloutbox}{red!70!black}{Critical Band (CB)}
For $N$ sinusoids \textit{close} to frequency $f_1$, the audibility condition is:
\[
\sum_i \sigma_i^2 > S_a(f_1)
\]
(energy integrates within the critical band)

Sinusoids are "near" if they fall within the \textbf{Critical Band} centered at $f_1$. CB width is a function of $f_1$.
\end{calloutbox}
\begin{center}\includegraphics[width=0.806\linewidth]{Pasted_image_20260501163938.png}\end{center}

\subsection{Masking Curves}

\subsubsection{Frequency Masking}

\textbf{Definition:} $S_m(f_0, \sigma^2, f)$ = minimum power of a tone at $f$ to be audible when a masker at $f_0$ with power $\sigma^2 > S_a(f_0)$ is playing.
\begin{center}\includegraphics[width=0.600\linewidth]{Pasted_image_20260501164014.png}\end{center}

\textbf{Properties of $S_m(f_0, \sigma^2, f)$}:
\begin{itemize}
\item Triangular shape around $f_0$; maximum at $f = f_0$
\item \textbf{Masking index}: $S_m(f, \sigma^2, f) - \sigma^2 < 0$ — masked tone need not be louder than masker
\item Asymmetric: slower decrease toward higher frequencies
\item Slope proportional to CB width
\item Slope (toward higher $f$) decreases as masker power $\sigma^2$ increases
\end{itemize}
\begin{center}\includegraphics[width=0.600\linewidth]{Pasted_image_20260522103757.png}\end{center}
\subsubsection{Time Masking}
\begin{center}\includegraphics[width=0.600\linewidth]{Pasted_image_20260501164647.png}\end{center}
\begin{itemize}
\item \textbf{Pre-masking}: 2–5 ms before onset of masker
\item \textbf{Post-masking}: 100–200 ms after masker ends
\end{itemize}

\subsection{Model Application}

Psychoacoustical model enables:
\begin{itemize}
\item Identify \textbf{non-audible signal components}
\item Allow quantization noise up to the masking threshold
\end{itemize}

\textbf{Limitations}:
\begin{itemize}
\item Model valid only for pure tones or narrowband sounds
\item Only 3-way simultaneous interactions modeled
\item Real signals are far more complex → parameters determined \textbf{experimentally} through large-scale listening tests
\end{itemize}

\subsection{YCbCr 4:2:0}

Most common format in video compression (JPEG, H.264, HEVC, AV1).

\begin{itemize}
\item \textbf{Y}: full resolution
\item \textbf{Cb, Cr}: half resolution horizontally and vertically
\end{itemize}

\textbf{Compression ratio} vs full RGB:
\[
\frac{4+1+1}{4+4+4} = \frac{6}{12} = 0.5 \quad \Rightarrow \quad 50\% \text{ data reduction}
\]

\textbf{Perceptual basis}: HVS is far more sensitive to luminance (brightness) variations than to chrominance (color) variations → reduced Cb/Cr resolution is largely imperceptible.

\subsection{Quantization}

\begin{itemize}
\item Samples represented on a discrete set: \textbf{uniform quantization} (rounding)
\item $L$ = number of levels; $b = \log_2 L$ bits per sample
\item Standard: $b = 8$ bits per component
\begin{itemize}
\item 256 gray levels (8 bpp)
\item $256^3 \approx 16\text{M}$ colors (24 bpp for RGB)
\end{itemize}
\item High Dynamic Range (HDR): 32–64 bits per channel
\end{itemize}

\section{Compression Principles}

\subsection{Motivation}

\begin{calloutbox}{green!55!black}{HD DVB Uncompressed Data Rate}
\textbf{System:} 1 luminance component $1920\times1080$ + 2 chrominance components $960\times540$, 8-bit quantization, 50 fps

\[
R = (1920\times1080 + 2\times960\times540)\times8\times50 \approx 1.24 \text{ Gbps}
\]

\textbf{2-hour movie storage:} $\approx 1.12$ TB

\textbf{Conclusion:} compression is essential.
\end{calloutbox}

\subsection{Compression Fundamentals}

Three sources of redundancy:

\begin{enumerate}
\item \textbf{Statistical redundancy}
\begin{itemize}
\item \textit{Spatial}: images are locally homogeneous (neighboring pixels are correlated)
\item \textit{Temporal}: successive video frames are highly correlated
\end{itemize}
\end{enumerate}

\begin{enumerate}
\item \textbf{Psychovisual redundancy}
\begin{itemize}
\item Limits of HVS: CSF, masking, contour sensitivity
\item Goal: \textbf{"Visually lossless"} coding — introduce errors below perception threshold
\end{itemize}
\end{enumerate}

\begin{enumerate}
\item \textbf{Semantic redundancy} \textit{(emerging)}
\begin{itemize}
\item Images contain recognizable structures (objects, concepts)
\item Goal: \textbf{"Generative" coding} — transmit tokens/meaning rather than pixels
\end{itemize}
\end{enumerate}

\textit{Modern algorithms should exploit all three.}

\subsection{Lossless and Lossy Algorithms}

\begin{center}
\begin{tabular}{|l|l|l|}
\hline
\textbf{Property} & \textbf{Lossless} & \textbf{Lossy} \\ \hline
Reconstruction & Perfect & Approximate \\ \hline
Basis & Statistical redundancy only & Quantization + psychovisual redundancy \\ \hline
Compression ratio & Small ($T \leq 3$) & High ($T \approx 5$ to $\gg$) \\ \hline
\end{tabular}
\end{center}

\subsection{Basic Tools for Compression}
\begin{center}\includegraphics[width=0.600\linewidth]{Pasted_image_20260501165037.png}\end{center}
\begin{enumerate}
\item \textbf{Transform}: Linear (DCT, Wavelet) or Non-linear (Neural) — concentrate energy into few coefficients
\item \textbf{Prediction}: Spatial or temporal redundancy reduction
\item \textbf{Quantization}: Rate reduction through approximation (\textbf{the only lossy step})
\item \textbf{Entropy Coding}: Residual redundancy reduction (VLC, Arithmetic coding)
\end{enumerate}

\subsection{Rate and Compression Ratio}

\textbf{Compression ratio}:
\[
T = \frac{B_{\text{in}}}{B_{\text{out}}} = \frac{R_{\text{in}}}{R_{\text{out}}}
\]

\textbf{Coding rate}:
\begin{itemize}
\item Image: $R = B_{\text{out}} / (NM)$ [bpp — bits per pixel]
\item Video/audio: $R = B_{\text{out}} / T$ [bps]
\end{itemize}

\begin{center}
\begin{tabular}{|l|l|}
\hline
\textbf{Application} & \textbf{Typical $T$} \\ \hline
Lossless image coding & $T \leq 3$ \\ \hline
Lossy image coding & $T \approx 5 \to ?$ \\ \hline
Lossy video coding & $T \approx 20 \to ?$ \\ \hline
\end{tabular}
\end{center}

\subsection{Quality and Distortion Metrics}

\subsubsection{Non-Perceptual Objective Criteria (NP-OC)}

\textbf{Error image}: $\mathcal{E}(f, \tilde{f}) = f - \tilde{f}$

\begin{calloutbox}{red!70!black}{Mean Square Error (MSE) and PSNR}
\[
\mathcal{D}(f,\tilde{f}) = \frac{1}{NM}\|\mathcal{E}\|^2 = \frac{1}{NM}\sum_{n,m}\mathcal{E}_{n,m}^2
\]

\[
\text{PSNR}(f,\tilde{f}) = 10\log_{10}\!\left(\frac{255^2}{\mathcal{D}(f,\tilde{f})}\right) \quad [\text{dB}]
\]

\textbf{Pros:} simple, analytically derivable, linked to $\mathcal{L}^2$ norm. \textbf{Cons:} does not model HVS — same MSE can yield very different perceived quality.
\end{calloutbox}

\subsubsection{Perceptual Objective Criteria}

\textbf{Weighted PSNR (WPSNR)}: apply a frequency weighting filter $h$ (modeling CSF):
\[
\text{WPSNR}(f,\tilde{f}) = 10\log_{10}\!\left(\frac{255^2}{\mathcal{D}_W}\right), \qquad \mathcal{D}_W = \frac{1}{NM}\|h * \mathcal{E}\|^2
\]
\begin{center}\includegraphics[width=0.600\linewidth]{Pasted_image_20260501165141.png}\end{center}
\begin{calloutbox}{red!70!black}{Structural Similarity Index (SSIM)}
For image blocks $x$ and $y$ (with $\alpha=\beta=\gamma=1$, $C_3=C_2/2$):
\[
\text{SSIM}(x,y) = \frac{(2\mu_x\mu_y + C_1)(2\sigma_{xy} + C_2)}{(\mu_x^2+\mu_y^2+C_1)(\sigma_x^2+\sigma_y^2+C_2)}
\]

Three components:
\begin{itemize}
\item \textbf{Luminance}: $l(x,y) = \frac{2\mu_x\mu_y + C_1}{\mu_x^2+\mu_y^2+C_1}$
\item \textbf{Contrast}: $c(x,y) = \frac{2\sigma_x\sigma_y + C_2}{\sigma_x^2+\sigma_y^2+C_2}$
\item \textbf{Structure}: $s(x,y) = \frac{\sigma_{xy}+C_3}{\sigma_x\sigma_y+C_3}$
\end{itemize}

Image SSIM = average over all blocks. Range $[0,1]$, 1 = identical.
\end{calloutbox}

\textbf{NN-based metrics (LPIPS)}:
\begin{itemize}
\item PSNR/SSIM fail to capture "realism" in highly compressed or AI-generated content
\item \textbf{LPIPS} (\textit{Learned Perceptual Image Patch Similarity}): uses pre-trained NNs (VGG, ViT) as feature extractors; measures distance in feature space rather than pixel space
\item Key insight: a signal can have low PSNR but high perceptual plausibility
\end{itemize}

\subsubsection{Subjective Criteria (SC)}

\begin{itemize}
\item Ground truth for quality evaluation: human observer experiments
\item Expensive, slow, statistically demanding
\item NP-OC often used as proxy (simplicity, geometric interpretation, analytical optimization) — but correlation with SC is imperfect
\end{itemize}

\subsection{Video Perception}

\begin{itemize}
\item HVS has both spatial and \textbf{temporal} sensitivity
\item \textbf{Spatio-temporal CSF}: sensitivity depends on both spatial frequency and temporal frequency
\item Spatial and temporal \textbf{masking} effects apply
\item \textbf{Scene changes} cause sudden sensitivity drops (masking → can tolerate more distortion just after a cut)
\end{itemize}

\subsection{Complexity, Delay and Robustness}

Three additional performance axes (beyond rate/quality):

\begin{itemize}
\item \textbf{Complexity}: limited by real-time constraints, hardware, economic cost
\item \textbf{Delay}: measured at encoder; related to complexity but mainly driven by \textbf{coding order} (how far ahead encoder must look)
\item \textbf{Robustness}: sensitivity of compressed stream to transmission losses
\end{itemize}

\textbf{Design tensions}:

\begin{center}
\begin{tabular}{|l|l|}
\hline
\textbf{Improve} & \textbf{Conflicts with} \\ \hline
↑ Quality & ↓ Rate \\ \hline
↑ Robustness & ↓ Complexity \\ \hline
— & ↓ Delay \\ \hline
\end{tabular}
\end{center}

Modern system design must balance all five: quality, rate, complexity, delay, robustness.

\textbf{Modern quality assessment approach}: report all three:
\begin{itemize}
\item \textbf{PSNR} (simple, pixel-based)
\item \textbf{SSIM} (structure-based)
\item \textbf{LPIPS} (feature-based, perceptual)
\end{itemize}
