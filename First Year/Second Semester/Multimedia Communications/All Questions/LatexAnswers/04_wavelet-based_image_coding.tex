\section{Wavelet-Based Image Coding}


\begin{questionbox}{Domanda 29}
State the time-frequency uncertainty principle ($\Delta t \cdot \Delta f \geq 1/4\pi$), explain why it imposes a trade-off, and how wavelets address it with adaptive multiresolution (short windows at high frequencies, long at low frequencies).
\end{questionbox}


$$
\Delta t \cdot \Delta f \geq \frac{1}{4\pi}
$$

\begin{itemize}
  \item \textbf{Short window} $\rightarrow$ good time/space localization, poor frequency resolution.
  \item \textbf{Long window} $\rightarrow$ good frequency resolution, poor localization.
  \item \textbf{Wavelets} $\rightarrow$ multiresolution:
  \begin{itemize}
    \item \textbf{high frequencies} $\rightarrow$ short basis functions;
    \item \textbf{low frequencies} $\rightarrow$ long basis functions.
  \end{itemize}
  \item \textbf{Image match} $\rightarrow$ edges need localization; smooth trends need coarse-scale representation.

\end{itemize}
\begin{questionbox}{Domanda 30}
Compare STFT (rigid tiling) and DWT (adaptive tiling) of the time-frequency plane, linking them to the trends vs anomalies image model.
\end{questionbox}


\begin{itemize}
  \item \textbf{STFT} $\rightarrow$ fixed window size $\rightarrow$ rigid time-frequency tiling; same resolution for all frequencies.
  \item \textbf{DWT} $\rightarrow$ filter banks + downsampling $\rightarrow$ low-frequency approximations + high-frequency detail subbands.
  \item \textbf{Repeated DWT} $\rightarrow$ multiresolution representation.
  \item \textbf{Image model}:
  \begin{itemize}
    \item \textbf{smooth trends} $\rightarrow$ coarse low-frequency components;
    \item \textbf{anomalies/edges} $\rightarrow$ high-frequency localized details.

  \end{itemize}
\end{itemize}

(TC\&JPEG) What is a primary advantage of the hierarchical (multiresolution) decomposition offered by the Wavelet Transform in JPEG2000?
\end{questionbox}


\begin{itemize}
  \item \textbf{Wavelet decomposition} $\rightarrow$ progressive and multiresolution coding.
  \item Low-resolution approximation decoded first $\rightarrow$ detail subbands refine quality and resolution.

\end{itemize}
\begin{questionbox}{Domanda 32}
(TC\&JPEG) Compare the block-based DCT approach used in JPEG with the wavelet-based decomposition used in JPEG2000, specifically in terms of how they handle the image signal and the resulting artifacts.

(TC\&JPEG) Which of the following is a key reason why JPEG2000 is generally more efficient than JPEG?
\end{questionbox}


\begin{itemize}
  \item \textbf{JPEG}:
  \begin{itemize}
    \item independent $8\times8$ block DCT;
    \item simple and efficient;
    \item \textbf{low bitrate} $\rightarrow$ \textbf{blocking artifacts}.
  \end{itemize}
  \item \textbf{JPEG2000}:
  \begin{itemize}
    \item DWT over large tiles / whole image;
    \item multiresolution representation;
    \item embedded bitplane coding and precise rate control;
    \item \textbf{low bitrate} $\rightarrow$ ringing/blurring near edges, not blocking.
  \end{itemize}
  \item \textbf{Efficiency reasons}:
  \begin{itemize}
    \item wavelet multiresolution decomposition;
    \item embedded bitstream truncation;
    \item independent codeblocks;
    \item quality/resolution scalability;
    \item no fixed $8\times8$ boundaries.

  \end{itemize}
\end{itemize}

\begin{questionbox}{Domanda 34}
Compare JPEG and JPEG2000 on channel-error robustness (codeblock independence, contained vs catastrophic propagation, resynchronization markers).
\end{questionbox}


\begin{itemize}
  \item \textbf{JPEG}:
  \begin{itemize}
    \item sequential entropy coding;
    \item bit error can desynchronize Huffman stream;
    \item corruption propagates until restart/resynchronization marker.
  \end{itemize}
  \item \textbf{JPEG2000}:
  \begin{itemize}
    \item relatively independent codeblocks;
    \item stream organized into packets/layers;
    \item damage tends to remain local to codeblock/subband/quality layer.
  \end{itemize}
  \item \textbf{Result} $\rightarrow$ JPEG2000 more robust to localized channel errors.

\end{itemize}
