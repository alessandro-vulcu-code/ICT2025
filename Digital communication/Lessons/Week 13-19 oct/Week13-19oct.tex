\subsection{Entropy}

\textbf{Entropy of a discrete-time random variable}
\begin{equation*}
\begin{aligned}
    \mathcal{H}(x) &= -\sum_{x} p_x(x) \log_2 p_x(x) \\
                   &= -\mathbb{E}[\log_2 p_x(x)].
\end{aligned}
\end{equation*}

\textbf{Joint entropy}
\begin{equation*}
\begin{aligned}
    \mathcal{H}(x_0, x_1) &= -\sum_{x_0}\sum_{x_1} p_{x_0 x_1}(x_0, x_1) \log_2 p_{x_0 x_1}(x_0, x_1) \\
                          &= -\mathbb{E}[\log_2 p_{x_0 x_1}(x_0, x_1)].
\end{aligned}
\end{equation*}

\begin{quote}
    \textbf{Nonnegative function that quantifies the amount of uncertainty in discrete valued RV x}
\end{quote}

\vspace{1em}

\textbf{Conditional entropy}
\begin{equation*}
    \mathcal{H}(x|y) = -\sum_{x}\sum_{y} p_{xy}(x,y) \log_2 p_{x|y}(x|y).
\end{equation*}

\textbf{Entropy connections}
\begin{equation*}
    \mathcal{H}(x,y) = \mathcal{H}(x) + \mathcal{H}(y|x).
\end{equation*}

\vspace{0.5cm}
The entropy is used in statistical physics.

\textbf{Joint entropy:}
\[
H(x_0, x_1) = \sum_{x_0} \sum_{x_1} p_{x_0, x_1}(x_0, x_1) \log_{1/2} p_{x_0, x_1}(x_0, x_1)
\]
% Nota: Nella formula manoscritta la base è 1/2, il che rimuove il segno meno.

\textbf{$\triangle$ Diff. inform. \underline{can} be negative}

\vspace{1em}

\textbf{Conditional entropy:}

% Nota: Trascrizione letterale della formula manoscritta.
% L'autore ha scritto H(x,y) a sinistra (invece di H(x|y)) e log_2 a destra senza segno meno esplicito.
\[
H(x,y) = \sum_{x} \sum_{y} p_{x,y}(x,y) \log_2 (p_{x|y}(x|y))
\]

From this, we can use the Bayes theorem to derive:

\begin{align*}
    H(x,y) &= H(x|y) + H(y) \\
           &= H(y|x) + H(x)
\end{align*}

\subsubsection{Proper complex}

A random variable (or a random process/random vector) is said to be proper complex if it satisfies \textbf{Properness}:

\vspace{0.5em}
\textbf{Properness}

\vspace{0.5em}
A complex random variable is said to be \textbf{proper} if
\[
E[x^2] = E[x]^2
\]
holds.
Properness requires $\mathrm{Re}[x], \mathrm{Im}[x]$ to be uncorrelated and have the same variance.

\vspace{0.8em}
A complex random vector is said to be \textbf{proper} if
\[
E[\mathbf{x}\mathbf{x}^T] = E[\mathbf{x}]E[\mathbf{x}]^T
\]
holds, which is equal to asking that $\mathrm{Re}[\boldsymbol{x}], \mathrm{Im}[\boldsymbol{x}]$ have identical Covariance matrices and cross-covariance equal to 0.

\vspace{0.8em}
Properness is preserved in affine transformations!

\subsection{Entropy rate}

Let's say we have a sequence of discrete random variables indexed by time, $x_0, \dots, x_{N-1}$.
If they are \textbf{IID}, then the entropy of the whole process grows linearly with $N$ at a rate $\mathcal{H}(x_0)$.

\vspace{0.5em}
\noindent
\textbf{In general}, a process grows according to the so-called \textit{entropy rate}
\begin{equation}
    \mathcal{H} = \lim_{N \to +\infty} \frac{1}{N} \mathcal{H}(x_0, \dots, x_{N-1}). \tag{1.32}
\end{equation}

\noindent
If the process is \textbf{stationary}, we can show that the entropy rate is equal to
\begin{equation}
    \mathcal{H} = \lim_{N \to +\infty} \mathcal{H}(x_N | x_0, \dots, x_{N-1}). \tag{1.33}
\end{equation}

\noindent
All of this can be extended for continuous processes to differential entropy.

\vspace{0.5em}
\noindent
A process is said to be \textbf{nonregular} if its present value is perfectly predictable from noiseless observations of the entire past, while the process is \textbf{regular} (which happens if $\mathfrak{h} > -\infty$) if its present value cannot be perfectly predicted from noiseless observations of the entire past.

\subsection{Differential entropy}

\textit{Differential entropy:}
\begin{equation*}
    h(x) = \int f_x(x) \log_{1/2} f_x(x) \, dx = \mathbb{E}\left[\log_{1/2} f_x(x)\right]
\end{equation*}

\textit{For vectors:}
\begin{equation*}
    h(\underline{x}) = \mathbb{E}\left[\log_{1/2} f_{\underline{x}}(\underline{x})\right]
\end{equation*}

The differential entropy rate does not measure the information contained in a random variable.
If we only have a random variable, we only have a shot. In order to consider a \textbf{rate}, we consider a \textbf{random process}: in this case, we have a random variable every certain amount of time: we can say that we have a certain amount of information per second!

A binary random process with $p=0.5$ which gives a value every second, gives 1 bit/s.

So the rate does not measure the information contained in a random variable.
We also need to keep in mind that the differential entropy can be negative.

\noindent\rule{\textwidth}{0.4pt}

It's easy to prove that a \textbf{Gaussian distribution maximizes the differential entropy}, for a fixed variance.

\noindent\rule{\textwidth}{0.4pt}

The differential entropy of a $b$ bit quantization of a random variable $x$ is $h(x)+b$

\textbf{Example 1.6 (Differential entropy of a complex Gaussian vector)}

\noindent
Let $\mathbf{x} \sim \mathcal{N}_{\mathcal{C}}(\boldsymbol{\mu}, \boldsymbol{R})$. From (C.15) and (1.22).
\begin{flalign*}
    h(\mathbf{x}) &= -\mathbb{E}[\log_2 f_{\mathbf{x}}(\mathbf{x})] & (1.23) \\
    &= \log_2 \det(\pi \boldsymbol{R}) + \mathbb{E}[(\mathbf{x} - \boldsymbol{\mu})^* \boldsymbol{R}^{-1} (\mathbf{x} - \boldsymbol{\mu})] \log_2 e & (1.24) \\
    &= \log_2 \det(\pi \boldsymbol{R}) + \text{tr}(\mathbb{E}[(\mathbf{x} - \boldsymbol{\mu})^* \boldsymbol{R}^{-1} (\mathbf{x} - \boldsymbol{\mu})]) \log_2 e & (1.25) \\
    &= \log_2 \det(\pi \boldsymbol{R}) + \text{tr}(\mathbb{E}[\boldsymbol{R}^{-1}(\mathbf{x} - \boldsymbol{\mu})(\mathbf{x} - \boldsymbol{\mu})^*]) \log_2 e & (1.26) \\
    &= \log_2 \det(\pi \boldsymbol{R}) + \text{tr}(\boldsymbol{R}^{-1}\underbrace{\mathbb{E}[(\mathbf{x} - \boldsymbol{\mu})(\mathbf{x} - \boldsymbol{\mu})^*]}_{\boldsymbol{R}}) \log_2 e & (1.27) \\
    &= \log_2 \det(\pi \boldsymbol{R}) + \text{tr}(\boldsymbol{I}) \log_2 e & (1.28) \\
    &= \log_2 \det(\pi e \boldsymbol{R}), & (1.29)
\end{flalign*}

\noindent
where in (1.25) we used the fact that a \underline{scalar equals its trace}, while in (1.26) we invoked the commutative property in (B.26).

    &= -\mathbb{E}\left[ \log_2 \left( \frac{1}{\det(\pi \boldsymbol{R}_x)} \exp(-(\boldsymbol{X} - \boldsymbol{\mu}_x)^* \boldsymbol{R}_x^{-1} (\boldsymbol{X} - \boldsymbol{\mu}_x)) \right) \right] \\
    &= \log_2(\det(\pi \boldsymbol{R}_x)) - \mathbb{E}\left[ \frac{\ln(\exp(-(\boldsymbol{X} - \boldsymbol{\mu}_x)^* \boldsymbol{R}_x^{-1} (\boldsymbol{X} - \boldsymbol{\mu}_x)))}{\ln 2} \right] \\
    &= \log_2(\det(\pi \boldsymbol{R}_x)) - \mathbb{E}\left[ \frac{-(\boldsymbol{X} - \boldsymbol{\mu}_x)^* \boldsymbol{R}_x^{-1} (\boldsymbol{X} - \boldsymbol{\mu}_x)}{\ln 2} \right] \\
    &= \log_2(\det(\pi \boldsymbol{R}_x)) + \mathbb{E}\left[ (\boldsymbol{X} - \boldsymbol{\mu}_x)^* \boldsymbol{R}_x^{-1} (\boldsymbol{X} - \boldsymbol{\mu}_x) \right] \log_2 e \tag{1.24}
\end{align*}

\noindent
and now we observe that $(\boldsymbol{X} - \boldsymbol{\mu}_x)^* \boldsymbol{R}_x^{-1} (\boldsymbol{X} - \boldsymbol{\mu}_x)$ actually is a scalar (given that $\boldsymbol{x}$ is a $N \times 1$ vector, we're multiplying $(1 \times N) \cdot (N \times N) \cdot (N \times 1) \rightarrow 1$): we can then exploit the fact that a scalar is equal to its trace together with the fact that the trace is a scalar operation to get
% ---------------------------------------------------------
% PAGINA 1: Dimostrazione Matematica e Definizioni
% ---------------------------------------------------------

\noindent
\begin{align}
    &= \log_2(\det(\pi \boldsymbol{R}_x)) + \text{Tr}\left[ \mathbb{E}\left[ (\boldsymbol{X} - \boldsymbol{\mu}_x)^* \boldsymbol{R}_x^{-1} (\boldsymbol{X} - \boldsymbol{\mu}_x) \right] \right] \log_2 e \tag{1.25} \\
    &= \log_2(\det(\pi \boldsymbol{R}_x)) + \mathbb{E}\left[ \text{Tr}\left[ (\boldsymbol{X} - \boldsymbol{\mu}_x)^* \boldsymbol{R}_x^{-1} (\boldsymbol{X} - \boldsymbol{\mu}_x) \right] \right] \log_2 e \qquad \text{Linearity} \nonumber \\
    &= \log_2(\det(\pi \boldsymbol{R}_x)) + \mathbb{E}\left[ \text{Tr}\left[ \boldsymbol{R}_x^{-1} (\boldsymbol{X} - \boldsymbol{\mu}_x)(\boldsymbol{X} - \boldsymbol{\mu}_x)^* \right] \right] \log_2 e \qquad \text{Ciclicity} \nonumber \\
    &= \log_2(\det(\pi \boldsymbol{R}_x)) + \text{Tr}\left[ \boldsymbol{R}_x^{-1} \boldsymbol{R}_x \right] \log_2 e \nonumber \\
    &= \log_2(\det(\pi \boldsymbol{R}_x)) + \text{Tr}\left[ \boldsymbol{I} \right] \log_2 e \tag{1.28}
\end{align}

\noindent
if then we consider $\boldsymbol{R}_x, \boldsymbol{I}$ as $N \times N$ matrices (which they are, this is just needed to say that $N$ is their measure), we'll have
\begin{align*}
    &= \log_2(\det(\pi \boldsymbol{R}_x)) + N \log_2 e \\
    &= \log_2(e^N \det(\pi \boldsymbol{R}_x))
\end{align*}
and, finally, for a matrix $\boldsymbol{A}$ of dimensionality $N \times N$ it holds that
\[
a^N \det(\boldsymbol{A}) = \det(a\boldsymbol{A}),
\]
so in our situation this is
\begin{equation}
    = \log_2(\det(\pi e \boldsymbol{R}_x)). \tag{1.29}
\end{equation}

\[
h(x) = \log_2(\pi e \sigma^2).

\subsection{Information divergence, mutual information}

We have that the Kullback-Leibler divergence is defined as
\begin{equation}
    \mathcal{D}(p||q) = 
    \begin{cases}
        \sum\limits_{x} p(x) \log_2 \left( \frac{p(x)}{q(x)} \right) & \text{discrete } x \\
        \int p(x) \log_2 \left( \frac{p(x)}{q(x)} \right) & \text{continuous } x
    \end{cases}
    \tag{1.34+1.36}
\end{equation}
this is also called the \textit{relative entropy}.

Another way to measure such thing is the Earth-Mover distance.

\subsection{Mutual information of a Gaussian scalar through an AWGN channel}
We were calculating the \textbf{mutual information} for a Gaussian complex scalar.

\vspace{0.5em}
\noindent
We talked about the relation $y = \sqrt{\rho}s + z$. We have that $s, z$ are \textbf{independent}.
It might seem weird that we're putting noise in a channel, but it will make sense later.

\vspace{0.5em}
\noindent
Our constellation input always will have $\sigma^2 = 1$, such that $s \sim \mathcal{N}_{\mathcal{C}}(0, 1)$; we'll also have a noise $z \sim \mathcal{N}_{\mathcal{C}}(0, 1)$. We have that $y$ is made of a sum of Gaussian independent random variables, and will thus be a Gaussian random variable, with a variance given by a sum of variances: we'll have

\begin{itemize}
    \item $y \sim \mathcal{N}_{\mathcal{C}}(0, 1 + \rho)$
    \item $y|s \sim \mathcal{N}_{\mathcal{C}}(\sqrt{\rho}s, 1)$: this is $y$ with $s$ being fixed; we fix the $\sqrt{\rho}s$ in $y$'s equation. We now want to calculate the mutual information between the input and the output of the signal.
\end{itemize}
\noindent
We know the differential entropy of a Gaussian scalar and we know that
\begin{flalign}
    \mathcal{I}(\rho) &= I(s; \sqrt{\rho}s + z) \nonumber \\
    &= h(\sqrt{\rho}s + z) - h(\sqrt{\rho}s + z | s) \nonumber \\
    &= h(\sqrt{\rho}s + z) - h(z) \nonumber \\
    &= \log_2(\pi e (1 + \rho)) - \log_2(\pi e) \nonumber \\
    &= \log_2(1 + \rho). \tag{1.51}
\end{flalign}

\noindent
\textbf{Relevant comments:}
\begin{itemize}
    \item In $h(\sqrt{\rho}s + z|s) = h(z)$, we know that we've already \textbf{fixed $s$}; $s$ multiplied by whichever scalar won't bring any extra information than $z$, as we already know what $s$ is; for this reason, the differential entropies are the same.
\end{itemize}

\vspace{0.5em}
\noindent
\textbf{If we then consider $\rho \to 0$}, we have

% Formula manoscritta per espansione di Taylor di log(1+x)
\vspace{1em}
\begin{center}
    $\log_e(1+x) \xrightarrow{x \to 0} x - \frac{1}{2} x^2 + \mathcal{O}(x^3)$
\end{center}
\vspace{1em}

\noindent
So we need to \textbf{convert to base $e$} and we'll then be able to use the expansion:

% Formula manoscritta per l'espansione in base 2
\vspace{1em}
\begin{center}
    $\log_2(1 + \rho) = \left(\rho - \frac{1}{2} \rho^2\right) \log_2 e + \mathcal{O}(\rho^3)$
\end{center}
\vspace{1em}

\noindent
And \textbf{for large $\rho$} we'll then have

% Formula manoscritta per l'approssimazione a infinito
\vspace{1em}
\begin{center}
     $\log_2(1 + \rho) \xrightarrow{\rho \to \infty} \log_2(\rho) + \mathcal{O}(1/\rho)$
\end{center}
\vspace{1em}
